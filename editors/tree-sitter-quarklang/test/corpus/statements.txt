fn main(IOStream io) void {
    int i = 0;
    if (i > 0) {
        io.println("pos");
    } else {
        io.println("non-pos");
    }
    while (i < 3) {
        i = i + 1;
        if (i == 2) {
            break;
        }
    }
    for (int j = 0; j < 3; j = j + 1) {
        io.println(j);
    }
    List<int> l = [1, 2, 3];
    for (int v : l) {
        io.println(v);
    }
    try {
        io.println(1 / 0);
    } catch (void e) {
        io.println("caught");
    }
    log i;
    delete l;
}
