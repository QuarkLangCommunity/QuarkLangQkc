/* QuarkLang user-space thread runtime (plan B: user-space API + parallel backend)
 * channel: mutex+cond FIFO (capacity 1024)
 * task: task object (fn + arg + done flag), executed by the worker thread pool
 * Cross-platform: POSIX pthread (Linux/macOS); the Windows branch uses _beginthreadex/WaitForSingleObject
 */
#ifdef _WIN32
/* Windows backend: _beginthreadex + SRWLOCK + CONDITION_VARIABLE (user-space API unchanged) */
#include <process.h>
#include <windows.h>
typedef struct { int cap, head, tail, cnt; int* buf; SRWLOCK mu; CONDITION_VARIABLE cv; } ql_channel;
typedef struct { void (*fn)(long long, long long, long long, long long, long long, long long, long long, long long); long long args[8]; volatile int done; SRWLOCK mu; } ql_task;
ql_channel* ql_channel_new(int cap) {
    ql_channel* c = calloc(1, sizeof(ql_channel));
    if (cap <= 0) cap = 1024;
    c->cap = cap; c->buf = malloc(sizeof(int) * cap);
    InitializeSRWLock(&c->mu); InitializeConditionVariable(&c->cv);
    return c;
}
int ql_send(ql_channel* c, int v) {
    AcquireSRWLockExclusive(&c->mu);
    while (c->cnt >= c->cap) SleepConditionVariableSRW(&c->cv, &c->mu, INFINITE, 0);
    c->buf[(c->head + c->cnt) % c->cap] = v; c->cnt++;
    WakeAllConditionVariable(&c->cv); ReleaseSRWLockExclusive(&c->mu);
    return 0;
}
int ql_recv(ql_channel* c) {
    AcquireSRWLockExclusive(&c->mu);
    while (c->cnt <= 0) SleepConditionVariableSRW(&c->cv, &c->mu, INFINITE, 0);
    int v = c->buf[c->head]; c->head = (c->head + 1) % c->cap; c->cnt--;
    WakeAllConditionVariable(&c->cv); ReleaseSRWLockExclusive(&c->mu);
    return v;
}
/* Thread handle: pid = small integer id (the compiler represents thread variables as i32, matching the language pid) */
#define QMAXTASK 4096
static ql_task* qtasks[QMAXTASK]; static int qtasks_n = 0;
int ql_spawn(void) {
    if (qtasks_n >= QMAXTASK - 1) return 0;
    ql_task* t = calloc(1, sizeof(ql_task));
    InitializeSRWLock(&t->mu);
    t->done = 1; /* a new thread = idle (the interpreter done semantics) */
    int id = ++qtasks_n; qtasks[id] = t; return id;
}
static unsigned __stdcall runner_w(void* p) { ql_task* t = p; t->fn(t->args[0], t->args[1], t->args[2], t->args[3], t->args[4], t->args[5], t->args[6], t->args[7]); AcquireSRWLockExclusive(&t->mu); t->done = 1; ReleaseSRWLockExclusive(&t->mu); return 0; }
void ql_merge(int id, void* fn, long long a0, long long a1, long long a2, long long a3, long long a4, long long a5, long long a6, long long a7) {
    if (id <= 0 || id >= QMAXTASK) return;
    ql_task* t = qtasks[id]; if (!t) return;
    t->fn = fn; t->args[0] = a0; t->args[1] = a1; t->args[2] = a2; t->args[3] = a3;
    t->args[4] = a4; t->args[5] = a5; t->args[6] = a6; t->args[7] = a7; t->done = 0;
    _beginthreadex(0, 0, runner_w, t, 0, 0);
}
void ql_block(int id) { if (id <= 0 || id >= QMAXTASK) return; ql_task* t = qtasks[id]; if (!t) return; while (1) { AcquireSRWLockExclusive(&t->mu); int d = t->done; ReleaseSRWLockExclusive(&t->mu); if (d) break; } }
int ql_done(int id) { if (id <= 0 || id >= QMAXTASK) return 1; ql_task* t = qtasks[id]; if (!t) return 1; AcquireSRWLockExclusive(&t->mu); int d = t->done; ReleaseSRWLockExclusive(&t->mu); return d; }

#include <pthread.h>

typedef struct { int cap, head, tail, cnt; int* buf; pthread_mutex_t mu; pthread_cond_t cv; } ql_channel;
/* ---- Thread pool (8 fixed worker threads + task queue; amortized, no per-task thread-creation cost) ---- */
#define QPOOL_N 8
#define QCAP 65536
typedef struct { void (*fn)(long long, long long, long long, long long, long long, long long, long long, long long); long long args[8]; volatile int done; } ql_task;   /* done made atomic: avoids a per-task mutex */
/* Lock-free MPSC ring queue: monotonic head/tail counters (atomic), zero locks on the fast path;
   workers spin briefly then sleep on a lock+condition; a producer signals only when someone is asleep — eliminating 8-core mutex ping-pong and per-task futex wakeups */
#define QMASK (QCAP - 1)
static ql_task* qq[QCAP];
static int qh, qt, qn;                 /* queue pointers / qn are protected by the ticket spinlock */
static unsigned qlk;                   /* ticket lock: 0 = free, taken with exchange, spins with pause (no futex syscall) */
static pthread_mutex_t qsmu = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t qscv = PTHREAD_COND_INITIALIZER;
static pthread_t qpool[QPOOL_N]; static int qpool_started = 0;

static void qlock(void) {
    while (__atomic_exchange_n(&qlk, 1, __ATOMIC_ACQUIRE)) {
        /* Cross-platform: the pause instruction is dispatched per architecture (arm64 has no __builtin_ia32_pause;
           CI on macOS (Apple Silicon) reports unknown builtin in practice) */
#if defined(__x86_64__) || defined(__i386__)
        __builtin_ia32_pause();
#elif defined(__aarch64__) || defined(__arm__)
        __asm__ __volatile__("yield");
#else
        /* Other architectures: not required */
#endif
    }
}
static void qunlock(void) {
    __atomic_store_n(&qlk, 0, __ATOMIC_RELEASE);
}

static void qwake(void) {
    pthread_mutex_lock(&qsmu); pthread_cond_signal(&qscv); pthread_mutex_unlock(&qsmu);
}

static void qpush(ql_task* t) {
    for (;;) {
        qlock();
        if (qn < QCAP) {
            int wasEmpty = (qn == 0);
            qq[qt] = t; qt = (qt + 1) & QMASK; qn++;
            qunlock();
            if (wasEmpty) qwake();
            return;
        }
        qunlock();
        pthread_mutex_lock(&qsmu); pthread_cond_wait(&qscv, &qsmu); pthread_mutex_unlock(&qsmu);
    }
}

static ql_task* qpop(void) {
    for (;;) {
        qlock();
        if (qn > 0) {
            ql_task* t = qq[qh]; qh = (qh + 1) & QMASK; qn--;
            qunlock();
            return t;
        }
        qunlock();
        pthread_mutex_lock(&qsmu); pthread_cond_wait(&qscv, &qsmu); pthread_mutex_unlock(&qsmu);
    }
}

static void* qworker(void* p) {
    for (;;) {
        ql_task* t = qpop();
        t->fn(t->args[0], t->args[1], t->args[2], t->args[3],
              t->args[4], t->args[5], t->args[6], t->args[7]);
        __atomic_store_n(&t->done, 1, __ATOMIC_RELEASE);
    }
    return 0;
}
static void qpool_ensure(void) {
    if (qpool_started) return;
    for (int i = 0; i < QPOOL_N; i++) pthread_create(&qpool[i], 0, qworker, 0);
    qpool_started = 1;
}
/* Channel: mutex + cond bounded FIFO (capacity cap; same semantics as the Windows branch) */
ql_channel* ql_channel_new(int cap) {
    ql_channel* c = calloc(1, sizeof(ql_channel));
    if (cap <= 0) cap = 1024;
    c->cap = cap; c->buf = malloc(sizeof(int) * cap);
    pthread_mutex_init(&c->mu, 0); pthread_cond_init(&c->cv, 0);
    return c;
}
int ql_send(ql_channel* c, int v) {
    pthread_mutex_lock(&c->mu);
    while (c->cnt >= c->cap) pthread_cond_wait(&c->cv, &c->mu);
    c->buf[(c->head + c->cnt) % c->cap] = v; c->cnt++;
    pthread_cond_broadcast(&c->cv);
    pthread_mutex_unlock(&c->mu);
    return 0;
}
int ql_recv(ql_channel* c) {
    pthread_mutex_lock(&c->mu);
    while (c->cnt <= 0) pthread_cond_wait(&c->cv, &c->mu);
    int v = c->buf[c->head]; c->head = (c->head + 1) % c->cap; c->cnt--;
    pthread_cond_broadcast(&c->cv);
    pthread_mutex_unlock(&c->mu);
    return v;
}
/* Thread handle: pid = small integer id (the compiler represents thread variables as i32, matching the language pid) */
#define QMAXTASK 4096
static ql_task* qtasks[QMAXTASK]; static int qtasks_n = 0;
int ql_spawn(void) {
    if (qtasks_n >= QMAXTASK - 1) return 0;
    ql_task* t = calloc(1, sizeof(ql_task));
    t->done = 1; /* a new thread = idle (the interpreter done semantics) */
    int id = ++qtasks_n; qtasks[id] = t; return id;
}
void ql_merge(int id, void* fn, long long a0, long long a1, long long a2, long long a3, long long a4, long long a5, long long a6, long long a7) {
    if (id <= 0 || id >= QMAXTASK) return;
    ql_task* t = qtasks[id]; if (!t) return;
    qpool_ensure();
    t->fn = fn; t->args[0] = a0; t->args[1] = a1; t->args[2] = a2; t->args[3] = a3;
    t->args[4] = a4; t->args[5] = a5; t->args[6] = a6; t->args[7] = a7; t->done = 0;
    qpush(t);
}
void ql_block(int id) {
    if (id <= 0 || id >= QMAXTASK) return;
    ql_task* t = qtasks[id]; if (!t) return;
    while (__atomic_load_n(&t->done, __ATOMIC_ACQUIRE) == 0) { }
}
int ql_done(int id) {
    if (id <= 0 || id >= QMAXTASK) return 1;
    ql_task* t = qtasks[id]; if (!t) return 1;
    return __atomic_load_n(&t->done, __ATOMIC_ACQUIRE);
}

/* String concatenation: malloc + copy (String variables + operator) */
char* ql_strcat(const char* a, const char* b) {
    size_t la = strlen(a), lb = strlen(b);
    char* r = malloc(la + lb + 1);
    memcpy(r, a, la); memcpy(r + la, b, lb); r[la + lb] = 0;
    return r;
}

/* ================= String built-in methods (same semantics as the interpreter internal/lang/eval.go) =================
 * rune (UTF-8) semantics: size/charAt/substring are per rune; contains/startsWith/endsWith/indexOf
 * are per byte (same semantics as the Go strings package). Out-of-range / parse failure → the same runtime error wording as the interpreter, exiting 1. */
static void ql_err_at(const char* msg, int line) {
    fprintf(stderr, "error: %s at line %d\n", msg, line);
    fflush(stderr);
    exit(1);
}
static int ql_u8seq(unsigned char c) {
    if (c < 0x80) return 1;
    if ((c & 0xE0) == 0xC0) return 2;
    if ((c & 0xF0) == 0xE0) return 3;
    if ((c & 0xF8) == 0xF0) return 4;
    return 1;
}
int ql_str_size(const char* s) {
    int n = 0;
    for (const unsigned char* p = (const unsigned char*)s; *p; ) { p += ql_u8seq(*p); n++; }
    return n;
}
/* ql_u8_off: byte offset of the i-th rune (i == rune count → strlen; i beyond → -1) */
static int ql_u8_off(const char* s, int i) {
    int b = 0;
    for (int k = 0; k < i; k++) {
        if (!s[b]) return -1;
        b += ql_u8seq((unsigned char)s[b]);
    }
    return b;
}
int ql_str_contains(const char* s, const char* sub) { return strstr(s, sub) != 0; }
int ql_str_startswith(const char* s, const char* p) { size_t lp = strlen(p); return strncmp(s, p, lp) == 0; }
int ql_str_endswith(const char* s, const char* p) {
    size_t ls = strlen(s), lp = strlen(p);
    return lp <= ls && memcmp(s + ls - lp, p, lp) == 0;
}
int ql_str_indexof(const char* s, const char* sub) {
    const char* p = strstr(s, sub);
    return p ? (int)(p - s) : -1;
}
char* ql_str_sub(const char* s, int start, int end, int line) {
    int n = ql_str_size(s);
    if (start < 0 || end < start || end > n) {
        char msg[160];
        snprintf(msg, sizeof msg, "StringIndexOutOfBoundsError: substring(%d, %d) 越界 [0,%d]", start, end, n);
        ql_err_at(msg, line);
    }
    int b0 = ql_u8_off(s, start), b1 = ql_u8_off(s, end);
    int len = b1 - b0;
    char* r = malloc(len + 1);
    memcpy(r, s + b0, len); r[len] = 0;
    return r;
}
char* ql_str_charat(const char* s, int i, int line) {
    int n = ql_str_size(s);
    int b = ql_u8_off(s, i);
    if (i < 0 || b < 0 || s[b] == 0) {
        char msg[160];
        snprintf(msg, sizeof msg, "StringIndexOutOfBoundsError: charAt(%d) 越界 [0,%d)", i, n);
        ql_err_at(msg, line);
    }
    int sz = ql_u8seq((unsigned char)s[b]);
    char* r = malloc(sz + 1);
    memcpy(r, s + b, sz); r[sz] = 0;
    return r;
}
/* The whitespace set of Go unicode.IsSpace (including the Zs category and U+0085) */
static int ql_space_at(const char* s, int* sz) {
    unsigned char c = (unsigned char)s[0];
    if (c == ' ' || c == '\t' || c == '\n' || c == '\v' || c == '\f' || c == '\r') { *sz = 1; return 1; }
    if (c == 0xC2 && (unsigned char)s[1] == 0x85) { *sz = 2; return 1; } /* U+0085 */
    if (c == 0xC2 && (unsigned char)s[1] == 0xA0) { *sz = 2; return 1; } /* U+00A0 */
    if (c == 0xE1 && (unsigned char)s[1] == 0x9A && (unsigned char)s[2] == 0x80) { *sz = 3; return 1; } /* U+1680 */
    if (c == 0xE2 && (unsigned char)s[1] == 0x80) {
        unsigned char d = (unsigned char)s[2];
        if (d >= 0x80 && d <= 0x8A) { *sz = 3; return 1; } /* U+2000..U+200A */
        if (d == 0xA8 || d == 0xA9) { *sz = 3; return 1; } /* U+2028/2029 */
        if (d == 0xAF) { *sz = 3; return 1; }              /* U+202F */
    }
    if (c == 0xE2 && (unsigned char)s[1] == 0x81 && (unsigned char)s[2] == 0x9F) { *sz = 3; return 1; } /* U+205F */
    if (c == 0xE3 && (unsigned char)s[1] == 0x80 && (unsigned char)s[2] == 0x80) { *sz = 3; return 1; } /* U+3000 */
    return 0;
}
char* ql_str_trim(const char* s, int mode) { /* 0=both 1=left 2=right */
    int len = (int)strlen(s);
    int b = 0, e = len;
    if (mode != 2) {
        while (b < e) { int sz = 0; if (!ql_space_at(s + b, &sz) || b + sz > e) break; b += sz; }
    }
    if (mode != 1) {
        while (e > b) {
            int k = e - 1;
            while (k > b && ((unsigned char)s[k] & 0xC0) == 0x80) k--;
            int sz = 0;
            if (!ql_space_at(s + k, &sz)) break;
            e = k;
        }
    }
    int n = e - b;
    char* r = malloc(n + 1);
    memcpy(r, s + b, n); r[n] = 0;
    return r;
}
/* Case mapping for ASCII + Latin-1 + the basic Greek/Cyrillic ranges (other scripts are left as-is) */
static int ql_lower_rune(int cp) {
    if (cp >= 'A' && cp <= 'Z') return cp + 32;
    if (cp >= 0xC0 && cp <= 0xDE && cp != 0xD7) return cp + 32;
    if (cp >= 0x391 && cp <= 0x3A9) return cp + 32;
    if (cp >= 0x410 && cp <= 0x42F) return cp + 32;
    if (cp >= 0x400 && cp <= 0x40F) return cp + 80;
    return cp;
}
static int ql_upper_rune(int cp) {
    if (cp >= 'a' && cp <= 'z') return cp - 32;
    if (cp >= 0xE0 && cp <= 0xFE && cp != 0xF7) return cp - 32;
    if (cp >= 0x3B1 && cp <= 0x3C9) return cp - 32;
    if (cp >= 0x430 && cp <= 0x44F) return cp - 32;
    if (cp >= 0x450 && cp <= 0x45F) return cp - 80;
    return cp;
}
static int ql_u8dec(const char* s, int* cp) {
    unsigned char c = (unsigned char)s[0];
    if (c < 0x80) { *cp = c; return 1; }
    if ((c & 0xE0) == 0xC0) { *cp = ((c & 0x1F) << 6) | ((unsigned char)s[1] & 0x3F); return 2; }
    if ((c & 0xF0) == 0xE0) { *cp = ((c & 0x0F) << 12) | (((unsigned char)s[1] & 0x3F) << 6) | ((unsigned char)s[2] & 0x3F); return 3; }
    if ((c & 0xF8) == 0xF0) { *cp = ((c & 0x07) << 18) | (((unsigned char)s[1] & 0x3F) << 12) | (((unsigned char)s[2] & 0x3F) << 6) | ((unsigned char)s[3] & 0x3F); return 4; }
    *cp = c; return 1;
}
static int ql_u8enc(int cp, char* out) {
    if (cp < 0x80) { out[0] = (char)cp; return 1; }
    if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
    if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
    out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F)); out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
}
static char* ql_str_mapcase(const char* s, int upper) {
    size_t l = strlen(s);
    char* r = malloc(l * 4 + 4);
    size_t o = 0;
    for (size_t i = 0; i < l; ) {
        int cp = 0, sz = ql_u8dec(s + i, &cp);
        cp = upper ? ql_upper_rune(cp) : ql_lower_rune(cp);
        char tmp[4];
        int n = ql_u8enc(cp, tmp);
        memcpy(r + o, tmp, n); o += n; i += sz;
    }
    r[o] = 0;
    return r;
}
char* ql_str_lower(const char* s) { return ql_str_mapcase(s, 0); }
char* ql_str_upper(const char* s) { return ql_str_mapcase(s, 1); }
char* ql_str_replace(const char* s, const char* old, const char* nw) {
    size_t lo = strlen(old), ln = strlen(nw);
    if (lo == 0) {
        size_t l = strlen(s), o = 0;
        char* r = malloc((l + 1) * (ln + 4) + 1);
        for (size_t i = 0; i < l; ) {
            memcpy(r + o, nw, ln); o += ln;
            int sz = ql_u8seq((unsigned char)s[i]);
            memcpy(r + o, s + i, sz); o += sz; i += sz;
        }
        memcpy(r + o, nw, ln); o += ln;
        r[o] = 0;
        return r;
    }
    size_t l = strlen(s), cap = l + 1, o = 0;
    char* r = malloc(cap);
    const char* p = s;
    for (;;) {
        const char* hit = strstr(p, old);
        size_t seg = hit ? (size_t)(hit - p) : strlen(p);
        if (o + seg + ln + 1 > cap) { cap = (o + seg + ln + 1) * 2; r = realloc(r, cap); }
        memcpy(r + o, p, seg); o += seg;
        if (!hit) break;
        memcpy(r + o, nw, ln); o += ln;
        p = hit + lo;
    }
    r[o] = 0;
    return r;
}
static void ql_parse_err(const char* s, const char* what, int line) {
    char msg[192];
    snprintf(msg, sizeof msg, "ParseError: \"%s\" 不是合法%s", s, what);
    ql_err_at(msg, line);
}
int ql_str_toint(const char* s, int line) {
    const char* p = s;
    while (*p == ' ' || *p == '\t' || *p == '\n' || *p == '\v' || *p == '\f' || *p == '\r') p++;
    const char* q = p + strlen(p);
    while (q > p && (q[-1] == ' ' || q[-1] == '\t' || q[-1] == '\n' || q[-1] == '\v' || q[-1] == '\f' || q[-1] == '\r')) q--;
    char buf[64];
    size_t n = (size_t)(q - p);
    if (n == 0 || n >= sizeof buf) ql_parse_err(s, "整数", line);
    memcpy(buf, p, n); buf[n] = 0;
    char* end = 0;
    long v = strtol(buf, &end, 10);
    if (end == buf || *end != 0 || v < -2147483648L || v > 2147483647L) ql_parse_err(s, "整数", line);
    return (int)v;
}
double ql_str_tofloat(const char* s, int line) {
    const char* p = s;
    while (*p == ' ' || *p == '\t' || *p == '\n' || *p == '\v' || *p == '\f' || *p == '\r') p++;
    const char* q = p + strlen(p);
    while (q > p && (q[-1] == ' ' || q[-1] == '\t' || q[-1] == '\n' || q[-1] == '\v' || q[-1] == '\f' || q[-1] == '\r')) q--;
    char buf[128];
    size_t n = (size_t)(q - p);
    if (n == 0 || n >= sizeof buf) ql_parse_err(s, "浮点数", line);
    memcpy(buf, p, n); buf[n] = 0;
    char* end = 0;
    double v = strtod(buf, &end);
    if (end == buf || *end != 0) ql_parse_err(s, "浮点数", line);
    return v;
}

/* ================= List built-in method helpers (int elements; same semantics as the interpreter value.go) ================= */
static int ql_cmp_int(const void* a, const void* b) {
    int x = *(const int*)a, y = *(const int*)b;
    return x < y ? -1 : (x > y ? 1 : 0);
}
void ql_list_int_sort(int* buf, int head, int tail) {
    if (tail - head > 1) qsort(buf + head, tail - head, sizeof(int), ql_cmp_int);
}
/* List.toString(): [e0, e1, ...] (only the visible range head..tail) */
char* ql_list_int_str(const int* buf, int head, int tail) {
    size_t cap = 16, o = 0;
    char* r = malloc(cap);
    r[o++] = '[';
    for (int i = head; i < tail; i++) {
        char nb[16];
        int n = snprintf(nb, sizeof nb, "%d", buf[i]);
        if (i > head) {
            if (o + 2 > cap) { cap *= 2; r = realloc(r, cap); }
            r[o++] = ','; r[o++] = ' ';
        }
        if (o + (size_t)n + 2 > cap) { cap = (o + n + 2) * 2; r = realloc(r, cap); }
        memcpy(r + o, nb, n); o += (size_t)n;
    }
    if (o + 2 > cap) { cap += 2; r = realloc(r, cap); }
    r[o++] = ']'; r[o] = 0;
    return r;
}

/* ================= Signature memorize runtime (@mb() memoization) =================
 * key = the argument list (int array), value = the wrapped function int return value;
 * semantics match the interpreter MemorizeBuffer: a hit returns directly, a miss executes and records. */
typedef struct ql_memo_ent { int n; int* keys; int val; struct ql_memo_ent* next; } ql_memo_ent;
typedef struct { ql_memo_ent** b; int cap; int len; } ql_memo;
static unsigned ql_memo_hash(const int* keys, int n) {
    unsigned h = 2166136261u; /* FNV-1a */
    for (int i = 0; i < n; i++) { h ^= (unsigned)keys[i]; h *= 16777619u; }
    return h;
}
void* ql_memo_new(void) {
    ql_memo* m = (ql_memo*)calloc(1, sizeof(ql_memo));
    m->cap = 64; m->len = 0;
    m->b = (ql_memo_ent**)calloc((size_t)m->cap, sizeof(ql_memo_ent*));
    return m;
}
static ql_memo_ent* ql_memo_find(ql_memo* m, const int* keys, int n) {
    ql_memo_ent* e = m->b[ql_memo_hash(keys, n) % (unsigned)m->cap];
    while (e) {
        if (e->n == n) {
            int i = 0;
            while (i < n && e->keys[i] == keys[i]) i++;
            if (i == n) return e;
        }
        e = e->next;
    }
    return 0;
}
int ql_memo_get(void* mp, int n, int* keys, int* found) {
    ql_memo* m = (ql_memo*)mp;
    ql_memo_ent* e = ql_memo_find(m, keys, n);
    *found = e ? 1 : 0;
    return e ? e->val : 0;
}
void ql_memo_put(void* mp, int n, int* keys, int val) {
    ql_memo* m = (ql_memo*)mp;
    if (ql_memo_find(m, keys, n)) return; /* already recorded: keep the first result (the interpreter Put overwrites, but same key means same value) */
    ql_memo_ent* e = (ql_memo_ent*)calloc(1, sizeof(ql_memo_ent));
    e->n = n; e->val = val;
    e->keys = (int*)malloc(sizeof(int) * (size_t)(n > 0 ? n : 1));
    for (int i = 0; i < n; i++) e->keys[i] = keys[i];
    unsigned h = ql_memo_hash(keys, n) % (unsigned)m->cap;
    e->next = m->b[h]; m->b[h] = e; m->len++;
}

/* ================= interface{} (tAny) runtime: RTTI descriptor dispatch =================
 * the descriptor layout must match the IR %RT = type { i8* (i8*)*, i8*, i32 }:
 *   { char* (*str)(void*); const char* name; int kind; }
 * kind: 0=int 1=float 2=bool 3=String 4=List<int> 5=struct 6=null */
typedef struct ql_rt { char* (*str)(void*); void* (*copy)(void*); const char* name; int kind; } ql_rt;
char* ql_int_to_str(int v);      /* IR helper generated by qkc (reuses the same formatting) */
char* ql_float_to_str(double v);
char* ql_long_to_str(long long v) {
    char* b = (char*)malloc(24);
    snprintf(b, 24, "%lld", v);
    return b;
}
char* ql_any_str_int(void* d) { return ql_int_to_str(*(int*)d); }
char* ql_any_str_float(void* d) { return ql_float_to_str(*(double*)d); }
char* ql_any_str_bool(void* d) { return (char*)(*(int*)d ? "true" : "false"); }
char* ql_any_str_string(void* d) { return *(char**)d; }
char* ql_any_str_list(void* d);
char* ql_any_str(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind == 6 || !d) return (char*)"nil";
    return r->str(d);
}
static void ql_any_bad(ql_rt* r, const char* want) {
    fprintf(stderr, "error: TypeError: cannot use interface{} value of type %s as %s\n",
            r ? r->name : "nil", want);
    exit(1);
}
int ql_any_int(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 0) ql_any_bad(r, "int");
    return *(int*)d;
}
double ql_any_float(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 1) ql_any_bad(r, "float");
    return *(double*)d;
}
int ql_any_bool(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 2) ql_any_bad(r, "bool");
    return *(int*)d;
}
char* ql_any_string(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 3) ql_any_bad(r, "String");
    return *(char**)d;
}
/* Equality semantics aligned with the interpreter equalValues: int/float compare numerically across types; bool/String by value;
 * List/struct by reference (pointer identity); nil == nil. */
int ql_any_eq(void* d1, void* rt1, void* d2, void* rt2) {
    ql_rt* a = (ql_rt*)rt1;
    ql_rt* b = (ql_rt*)rt2;
    int k1 = a ? a->kind : 6, k2 = b ? b->kind : 6;
    if (k1 == 6 || k2 == 6) return k1 == k2;
    if ((k1 == 0 || k1 == 1) && (k2 == 0 || k2 == 1)) {
        double x = (k1 == 0) ? (double)*(int*)d1 : *(double*)d1;
        double y = (k2 == 0) ? (double)*(int*)d2 : *(double*)d2;
        return x == y;
    }
    if (k1 != k2) return 0;
    switch (k1) {
    case 2: return (*(int*)d1 != 0) == (*(int*)d2 != 0);
    case 3: return strcmp(*(char**)d1, *(char**)d2) == 0;
    case 4: case 5: return d1 == d2;
    }
    return 0;
}
/* ---- deep copy of interface{} values (per RTTI; reused by HashTable.Put and copyd) ----
 * semantics aligned with the interpreter deepCopy: List copies the buffer, struct is not copied (pointer shared),
 * String/scalars are snapshotted by value (heap cell copied). */
void* ql_any_copy_int(void* d) { int* p = (int*)malloc(4); *p = *(int*)d; return p; }
void* ql_any_copy_float(void* d) { double* p = (double*)malloc(8); *p = *(double*)d; return p; }
void* ql_any_copy_bool(void* d) { int* p = (int*)malloc(4); *p = *(int*)d; return p; }
void* ql_any_copy_string(void* d) { char** p = (char**)malloc(8); *p = *(char**)d; return p; }
void* ql_any_copy_identity(void* d) { return d; }
void* ql_any_copy_nil(void* d) { (void)d; return 0; }



/* String.split(sep) → List<String>: same as Go strings.Split (empty sep → per UTF-8 character) */
static void ql_strlist_push(char*** buf, int* cap, int* n, const char* p, size_t len) {
    if (*n >= *cap) { *cap = *cap ? *cap * 2 : 4; *buf = (char**)realloc(*buf, sizeof(char*) * (size_t)(*cap)); }
    char* e = (char*)malloc(len + 1);
    memcpy(e, p, len); e[len] = 0;
    (*buf)[(*n)++] = e;
}
void* ql_str_split(const char* s, const char* sep) {
    char** buf = 0; int cap = 0, n = 0;
    size_t sn = strlen(sep);
    if (sn == 0) {
        const char* p = s;
        while (*p) {
            unsigned char c = (unsigned char)*p;
            int l = (c >= 0xF0) ? 4 : (c >= 0xE0) ? 3 : (c >= 0xC0) ? 2 : 1;
            ql_strlist_push(&buf, &cap, &n, p, (size_t)l);
            p += l;
        }
    } else {
        const char* p = s;
        for (;;) {
            const char* q = strstr(p, sep);
            if (!q) { ql_strlist_push(&buf, &cap, &n, p, strlen(p)); break; }
            ql_strlist_push(&buf, &cap, &n, p, (size_t)(q - p));
            p = q + sn;
        }
    }
    if (!buf) buf = (char**)malloc(sizeof(char*));
    struct ql_split_list { char** buf; int head; int tail; };
    struct ql_split_list* l = (struct ql_split_list*)calloc(1, sizeof(struct ql_split_list));
    l->buf = buf; l->head = 0; l->tail = n;
    return l;
}

/* List<String>: same layout as %ListS { i8** buf, i32 head, i32 tail }; toString same as the interpreter List.String() */
typedef struct { char** buf; int head; int tail; } ql_list_s;
char* ql_list_str_str(char** buf, int head, int tail) {
    size_t cap = 16, o = 0;
    char* r = (char*)malloc(cap);
    r[o++] = '[';
    for (int i = head; i < tail; i++) {
        const char* e = buf[i];
        size_t n = strlen(e);
        if (i > head) {
            if (o + 2 > cap) { cap *= 2; r = (char*)realloc(r, cap); }
            r[o++] = ','; r[o++] = ' ';
        }
        if (o + n + 2 > cap) { while (o + n + 2 > cap) cap *= 2; r = (char*)realloc(r, cap); }
        memcpy(r + o, e, n); o += n;
    }
    if (o + 2 > cap) { cap += 2; r = (char*)realloc(r, cap); }
    r[o++] = ']'; r[o] = 0;
    return r;
}

/* ================= HashTable<K,V> runtime =================
 * key = the interpreter-style structured string TypeName + ":" + String() (produced by ql_table_key);
 * value = the boxed interface{} value (data + RTTI descriptor), deep-copied via RTTI copy on Put. */
typedef struct ql_tent { char* key; void* data; void* rt; struct ql_tent* next; } ql_tent;
typedef struct { ql_tent** b; int cap; int len; } ql_table;
static unsigned ql_tbl_hash(const char* s) {
    unsigned h = 2166136261u;
    while (*s) { h ^= (unsigned char)*s++; h *= 16777619u; }
    return h;
}
void* ql_table_new(void) {
    ql_table* t = (ql_table*)calloc(1, sizeof(ql_table));
    t->cap = 64;
    t->b = (ql_tent**)calloc((size_t)t->cap, sizeof(ql_tent*));
    return t;
}
static ql_tent* ql_table_find(ql_table* t, const char* key) {
    ql_tent* e = t->b[ql_tbl_hash(key) % (unsigned)t->cap];
    while (e) { if (strcmp(e->key, key) == 0) return e; e = e->next; }
    return 0;
}
/* ql_table_key: turns a boxed key into the interpreter-style key string (TypeName:String()). */
char* ql_table_key(void* data, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    const char* tn = r ? r->name : "nil";
    const char* sv = (r && data) ? r->str(data) : "nil";
    size_t n = strlen(tn) + strlen(sv) + 2;
    char* k = (char*)malloc(n);
    snprintf(k, n, "%s:%s", tn, sv);
    return k;
}
static void ql_table_grow(ql_table* t);
void ql_table_put(void* tp, char* key, void* data, void* rt) {
    ql_table* t = (ql_table*)tp;
    ql_rt* r = (ql_rt*)rt;
    void* cp = r ? (r->copy ? r->copy(data) : data) : 0;
    ql_tent* e = ql_table_find(t, key);
    if (e) { e->data = cp; e->rt = rt; return; }
    e = (ql_tent*)calloc(1, sizeof(ql_tent));
    e->key = key; e->data = cp; e->rt = rt;
    unsigned h = ql_tbl_hash(key) % (unsigned)t->cap;
    e->next = t->b[h]; t->b[h] = e; t->len++;
    if (t->len > t->cap * 2) ql_table_grow(t);
}
static void ql_table_grow(ql_table* t) {
    int nc = t->cap * 2;
    ql_tent** nb = (ql_tent**)calloc((size_t)nc, sizeof(ql_tent*));
    for (int i = 0; i < t->cap; i++) {
        ql_tent* e = t->b[i];
        while (e) {
            ql_tent* nx = e->next;
            unsigned h = ql_tbl_hash(e->key) % (unsigned)nc;
            e->next = nb[h]; nb[h] = e;
            e = nx;
        }
    }
    free(t->b); t->b = nb; t->cap = nc;
}
void* ql_table_get(void* tp, char* key, void** out_rt) {
    ql_table* t = (ql_table*)tp;
    ql_tent* e = ql_table_find(t, key);
    if (!e) { if (out_rt) *out_rt = 0; return 0; }
    if (out_rt) *out_rt = e->rt;
    return e->data;
}
int ql_table_contains(void* tp, char* key) { return ql_table_find((ql_table*)tp, key) != 0; }
void ql_table_remove(void* tp, char* key) {
    ql_table* t = (ql_table*)tp;
    unsigned h = ql_tbl_hash(key) % (unsigned)t->cap;
    ql_tent** pp = &t->b[h];
    while (*pp) {
        if (strcmp((*pp)->key, key) == 0) { *pp = (*pp)->next; t->len--; return; }
        pp = &(*pp)->next;
    }
}
int ql_table_size(void* tp) { return ((ql_table*)tp)->len; }
/* Key list (deterministic order: strcmp ascending; the interpreter keys() uses Go map order, itself nondeterministic). */
typedef struct { char** buf; int head; int tail; } ql_strlist;
static int ql_cmp_cstr(const void* a, const void* b) {
    return strcmp(*(char* const*)a, *(char* const*)b);
}
void* ql_table_keys(void* tp) {
    ql_table* t = (ql_table*)tp;
    int n = t->len;
    char** buf = (char**)malloc(sizeof(char*) * (size_t)(n > 0 ? n : 1));
    int k = 0;
    for (int i = 0; i < t->cap; i++)
        for (ql_tent* e = t->b[i]; e; e = e->next) {
            /* Keys() returns the **original value** of the key: a String key drops the "String:" prefix (same rule as the interpreter Keys) */
            buf[k++] = (strncmp(e->key, "String:", 7) == 0) ? (e->key + 7) : e->key;
        }
    if (n > 1) qsort(buf, (size_t)n, sizeof(char*), ql_cmp_cstr);
    ql_strlist* l = (ql_strlist*)calloc(1, sizeof(ql_strlist));
    l->buf = buf; l->head = 0; l->tail = n;
    return l;
}


/* Concrete V lookup: after confirming non-nil, returns the object pointer (for struct/List; missing key → an explicit runtime error) */
void* ql_any_ptr(void* d, void* rt) {
    if (!rt || !d) {
        fprintf(stderr, "error: TypeError: HashTable.get: missing key (nil value)\n");
        exit(1);
    }
    return d;
}
/* RTTI deep copy for List<int> (the visible range is copied into a new buffer) */
typedef struct { int* buf; int head; int tail; } ql_list_c;
void* ql_any_copy_list(void* d) {
    ql_list_c* src = (ql_list_c*)d;
    if (!src) return 0;
    int n = src->tail - src->head;
    if (n < 0) n = 0;
    int* buf = (int*)malloc(sizeof(int) * (size_t)(n > 0 ? n : 1));
    for (int i = 0; i < n; i++) buf[i] = src->buf[src->head + i];
    ql_list_c* cp = (ql_list_c*)calloc(1, sizeof(ql_list_c));
    cp->buf = buf; cp->head = 0; cp->tail = n;
    return cp;
}

char* ql_any_str_list(void* d) {
    struct ql_list_s { int* buf; int head; int tail; };
    struct ql_list_s* l = (struct ql_list_s*)d;
    return ql_list_int_str(l->buf, l->head, l->tail);
}
