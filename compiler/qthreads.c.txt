/* QuarkLang user-space thread runtime (plan B: user-space API + parallel backend)
 * channel: mutex+cond FIFO (capacity 1024)
 * task: task object (fn + arg + done flag), executed by the worker thread pool
 * Cross-platform: POSIX pthread (Linux/macOS); the Windows branch uses _beginthreadex/WaitForSingleObject
 */
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

/* Set before the first worker thread exists. A program that has not started the task pool cannot
 * allocate from two threads, so the allocator skips its lock there (design R2's "no lock when
 * single-threaded"); the pool sets this to 1 before creating workers. */
static volatile int ql_mem_multithreaded;

#ifdef _WIN32
/* Windows backend: _beginthreadex + SRWLOCK + CONDITION_VARIABLE (user-space API unchanged) */
#include <process.h>
#include <windows.h>
typedef struct { int cap, head, tail, cnt; int* buf; SRWLOCK mu; CONDITION_VARIABLE cv; } ql_channel;
typedef struct { void (*fn)(long long, long long, long long, long long, long long, long long, long long, long long); long long args[8]; volatile int done; SRWLOCK mu; } ql_task;
ql_channel* ql_channel_new(int cap) {
    ql_channel* c = calloc(1, sizeof(ql_channel));
    if (cap <= 0) cap = 1024;
    c->cap = cap; c->buf = malloc(sizeof(int) * cap);
    InitializeSRWLock(&c->mu); InitializeConditionVariable(&c->cv);
    return c;
}
int ql_send(ql_channel* c, int v) {
    AcquireSRWLockExclusive(&c->mu);
    while (c->cnt >= c->cap) SleepConditionVariableSRW(&c->cv, &c->mu, INFINITE, 0);
    c->buf[(c->head + c->cnt) % c->cap] = v; c->cnt++;
    WakeAllConditionVariable(&c->cv); ReleaseSRWLockExclusive(&c->mu);
    return 0;
}
int ql_recv(ql_channel* c) {
    AcquireSRWLockExclusive(&c->mu);
    while (c->cnt <= 0) SleepConditionVariableSRW(&c->cv, &c->mu, INFINITE, 0);
    int v = c->buf[c->head]; c->head = (c->head + 1) % c->cap; c->cnt--;
    WakeAllConditionVariable(&c->cv); ReleaseSRWLockExclusive(&c->mu);
    return v;
}
/* Thread handle: pid = small integer id (the compiler represents thread variables as i32, matching the language pid) */
#define QMAXTASK 4096
static ql_task* qtasks[QMAXTASK]; static int qtasks_n = 0;
int ql_spawn(void) {
    if (qtasks_n >= QMAXTASK - 1) return 0;
    ql_task* t = calloc(1, sizeof(ql_task));
    InitializeSRWLock(&t->mu);
    t->done = 1; /* a new thread = idle (the interpreter done semantics) */
    int id = ++qtasks_n; qtasks[id] = t; return id;
}
static unsigned __stdcall runner_w(void* p) { ql_task* t = p; t->fn(t->args[0], t->args[1], t->args[2], t->args[3], t->args[4], t->args[5], t->args[6], t->args[7]); AcquireSRWLockExclusive(&t->mu); t->done = 1; ReleaseSRWLockExclusive(&t->mu); return 0; }
void ql_merge(int id, void* fn, long long a0, long long a1, long long a2, long long a3, long long a4, long long a5, long long a6, long long a7) {
    if (id <= 0 || id >= QMAXTASK) return;
    ql_task* t = qtasks[id]; if (!t) return;
    t->fn = fn; t->args[0] = a0; t->args[1] = a1; t->args[2] = a2; t->args[3] = a3;
    t->args[4] = a4; t->args[5] = a5; t->args[6] = a6; t->args[7] = a7; t->done = 0;
#if defined(__GNUC__) || defined(__clang__)
    /* This branch is not compiled by today's qkc (compiler/main.go returns no runtime source on
     * Windows), but if it is ever linked the allocator must stop taking its single-threaded path. */
    __atomic_store_n(&ql_mem_multithreaded, 1, __ATOMIC_RELEASE);
#endif
    _beginthreadex(0, 0, runner_w, t, 0, 0);
}
void ql_block(int id) { if (id <= 0 || id >= QMAXTASK) return; ql_task* t = qtasks[id]; if (!t) return; while (1) { AcquireSRWLockExclusive(&t->mu); int d = t->done; ReleaseSRWLockExclusive(&t->mu); if (d) break; } }
int ql_done(int id) { if (id <= 0 || id >= QMAXTASK) return 1; ql_task* t = qtasks[id]; if (!t) return 1; AcquireSRWLockExclusive(&t->mu); int d = t->done; ReleaseSRWLockExclusive(&t->mu); return d; }
#else
/* POSIX implementation (original) */
#include <pthread.h>
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

typedef struct { int cap, head, tail, cnt; int* buf; pthread_mutex_t mu; pthread_cond_t cv; } ql_channel;
/* ---- Thread pool (8 fixed worker threads + task queue; amortized, no per-task thread-creation cost) ---- */
#define QPOOL_N 8
#define QCAP 65536
typedef struct { void (*fn)(long long, long long, long long, long long, long long, long long, long long, long long); long long args[8]; volatile int done; } ql_task;   /* done made atomic: avoids a per-task mutex */
/* Lock-free MPSC ring queue: monotonic head/tail counters (atomic), zero locks on the fast path;
   workers spin briefly then sleep on a lock+condition; a producer signals only when someone is asleep — eliminating 8-core mutex ping-pong and per-task futex wakeups */
#define QMASK (QCAP - 1)
static ql_task* qq[QCAP];
static int qh, qt, qn;                 /* queue pointers / qn are protected by the ticket spinlock */
static unsigned qlk;                   /* ticket lock: 0 = free, taken with exchange, spins with pause (no futex syscall) */
static pthread_mutex_t qsmu = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t qscv = PTHREAD_COND_INITIALIZER;
static pthread_t qpool[QPOOL_N]; static int qpool_started = 0;

static void qlock(void) {
    while (__atomic_exchange_n(&qlk, 1, __ATOMIC_ACQUIRE)) {
        /* Cross-platform: the pause instruction is dispatched per architecture (arm64 has no __builtin_ia32_pause;
           CI on macOS (Apple Silicon) reports unknown builtin in practice) */
#if defined(__x86_64__) || defined(__i386__)
        __builtin_ia32_pause();
#elif defined(__aarch64__) || defined(__arm__)
        __asm__ __volatile__("yield");
#else
        /* Other architectures: not required */
#endif
    }
}
static void qunlock(void) {
    __atomic_store_n(&qlk, 0, __ATOMIC_RELEASE);
}

static void qwake(void) {
    pthread_mutex_lock(&qsmu); pthread_cond_signal(&qscv); pthread_mutex_unlock(&qsmu);
}

static void qpush(ql_task* t) {
    for (;;) {
        qlock();
        if (qn < QCAP) {
            int wasEmpty = (qn == 0);
            qq[qt] = t; qt = (qt + 1) & QMASK; qn++;
            qunlock();
            if (wasEmpty) qwake();
            return;
        }
        qunlock();
        pthread_mutex_lock(&qsmu); pthread_cond_wait(&qscv, &qsmu); pthread_mutex_unlock(&qsmu);
    }
}

static ql_task* qpop(void) {
    for (;;) {
        qlock();
        if (qn > 0) {
            ql_task* t = qq[qh]; qh = (qh + 1) & QMASK; qn--;
            qunlock();
            return t;
        }
        qunlock();
        pthread_mutex_lock(&qsmu); pthread_cond_wait(&qscv, &qsmu); pthread_mutex_unlock(&qsmu);
    }
}

static void* qworker(void* p) {
    for (;;) {
        ql_task* t = qpop();
        t->fn(t->args[0], t->args[1], t->args[2], t->args[3],
              t->args[4], t->args[5], t->args[6], t->args[7]);
        __atomic_store_n(&t->done, 1, __ATOMIC_RELEASE);
    }
    return 0;
}
static void qpool_ensure(void) {
    if (qpool_started) return;
    /* Store before pthread_create: a reader that still sees 0 is before any worker exists, and a
     * task cannot run until after qpush(), which is after this store. */
    __atomic_store_n(&ql_mem_multithreaded, 1, __ATOMIC_RELEASE);
    for (int i = 0; i < QPOOL_N; i++) pthread_create(&qpool[i], 0, qworker, 0);
    qpool_started = 1;
}
/* Channel: mutex + cond bounded FIFO (capacity cap; same semantics as the Windows branch) */
ql_channel* ql_channel_new(int cap) {
    ql_channel* c = calloc(1, sizeof(ql_channel));
    if (cap <= 0) cap = 1024;
    c->cap = cap; c->buf = malloc(sizeof(int) * cap);
    pthread_mutex_init(&c->mu, 0); pthread_cond_init(&c->cv, 0);
    return c;
}
int ql_send(ql_channel* c, int v) {
    pthread_mutex_lock(&c->mu);
    while (c->cnt >= c->cap) pthread_cond_wait(&c->cv, &c->mu);
    c->buf[(c->head + c->cnt) % c->cap] = v; c->cnt++;
    pthread_cond_broadcast(&c->cv);
    pthread_mutex_unlock(&c->mu);
    return 0;
}
int ql_recv(ql_channel* c) {
    pthread_mutex_lock(&c->mu);
    while (c->cnt <= 0) pthread_cond_wait(&c->cv, &c->mu);
    int v = c->buf[c->head]; c->head = (c->head + 1) % c->cap; c->cnt--;
    pthread_cond_broadcast(&c->cv);
    pthread_mutex_unlock(&c->mu);
    return v;
}
/* Thread handle: pid = small integer id (the compiler represents thread variables as i32, matching the language pid) */
#define QMAXTASK 4096
static ql_task* qtasks[QMAXTASK]; static int qtasks_n = 0;
int ql_spawn(void) {
    if (qtasks_n >= QMAXTASK - 1) return 0;
    ql_task* t = calloc(1, sizeof(ql_task));
    t->done = 1; /* a new thread = idle (the interpreter done semantics) */
    int id = ++qtasks_n; qtasks[id] = t; return id;
}
void ql_merge(int id, void* fn, long long a0, long long a1, long long a2, long long a3, long long a4, long long a5, long long a6, long long a7) {
    if (id <= 0 || id >= QMAXTASK) return;
    ql_task* t = qtasks[id]; if (!t) return;
    qpool_ensure();
    t->fn = fn; t->args[0] = a0; t->args[1] = a1; t->args[2] = a2; t->args[3] = a3;
    t->args[4] = a4; t->args[5] = a5; t->args[6] = a6; t->args[7] = a7; t->done = 0;
    qpush(t);
}
void ql_block(int id) {
    if (id <= 0 || id >= QMAXTASK) return;
    ql_task* t = qtasks[id]; if (!t) return;
    while (__atomic_load_n(&t->done, __ATOMIC_ACQUIRE) == 0) { }
}
int ql_done(int id) {
    if (id <= 0 || id >= QMAXTASK) return 1;
    ql_task* t = qtasks[id]; if (!t) return 1;
    return __atomic_load_n(&t->done, __ATOMIC_ACQUIRE);
}

/* String concatenation: malloc + copy (String variables + operator) */
char* ql_strcat(const char* a, const char* b) {
    size_t la = strlen(a), lb = strlen(b);
    char* r = malloc(la + lb + 1);
    memcpy(r, a, la); memcpy(r + la, b, lb); r[la + lb] = 0;
    return r;
}

/* ================= Counted-slab allocator (stage 1 of docs/compiled-memory-design.md) =================
 *
 * This is the compiled runtime's memory policy: a block is a slab of `ql_mem_block_size` bytes
 * dedicated to one size class; an allocation occupies one slot of that class; release is per object.
 * The classes are small enough that a List header, a list buffer, a boxed scalar and a struct share the
 * same path, while a request above the largest class gets a dedicated large block (one object, tail
 * waste counted by Fragmentation).
 *
 * The public allocation API keeps the design's owner argument even though stage 1 records the owner
 * rather than enforcing it (D5 is still open):
 *   ql_alloc(size, owner), ql_calloc(count, size, owner), ql_realloc(ptr, size), ql_free(ptr).
 *
 * The language surface is deliberate: stage 1 routes what cgen emits and what an owned `List<int>` can
 * release. The runtime's other C allocations stay on the host allocator until stage 3 decides D1; the
 * one exception is `ql_any_copy_list`, which can hand a `List<int>` to a deletable variable and is
 * routed below.
 *
 * D3 reading: release preserves the payload (it does not wipe). `ql_calloc` zeroes because callers
 * asked for calloc; `ql_alloc` does not. A later same-class allocation may therefore see the previous
 * payload; the standard's contradictory "never hands another value's bytes" sentence is still an owner
 * decision, recorded in docs/compiled-memory-stage1.md.
 *
 * D10 reading: every routed allocation is an opaque call. The compiler never elides one and LLVM cannot
 * see through it, which is what makes occupancy/fragmentation observable. The cost is measured, not
 * assumed (R1/R2).
 */
#ifndef QL_MEM_DEBUG
#define QL_MEM_DEBUG 0
#endif

#define QL_MEM_CLASS_COUNT 7
#define QL_MEM_ALIGN 16
#define QL_MEM_OBJ_HDR 32
#define QL_MEM_DEFAULT_BLOCK 4096
#define QL_MEM_MAX_BLOCK (1 << 20)
#define QL_MEM_OBJ_MAGIC 0x514C4F42u   /* "QLOB" */
#define QL_MEM_SLAB_MAGIC 0x514C5342u  /* "QLSB" */
#define QL_MEM_LARGE_MAGIC 0x514C4C42u /* "QLLB" */
#define QL_MEM_KIND_SLOT 1u
#define QL_MEM_KIND_LARGE 2u
#define QL_MEM_FLAG_LIVE 1u

static const size_t ql_mem_class_size[QL_MEM_CLASS_COUNT] = {16, 32, 64, 128, 256, 512, 1024};

typedef struct ql_slab ql_slab;
typedef struct ql_obj_hdr ql_obj_hdr;

/* One object's header. The payload starts at header + 32, so a 16-byte-aligned slab slot gives a
 * 16-byte-aligned payload. `next_free` lives here, never in the payload: release must preserve the
 * payload, so the allocator cannot overwrite its first bytes with a free-list pointer. */
struct ql_obj_hdr {
    uint32_t magic;
    uint16_t kind;
    uint16_t flags;
    size_t requested;
    void* block; /* ql_slab* for a slot, ql_large_meta* for a large object */
    ql_obj_hdr* next_free;
};

/* Metadata for a large block. It sits before the object header, so ql_free still finds the common
 * header at payload - 32 and reaches this metadata through h->block. */
typedef struct ql_large_meta {
    uint32_t magic;
    uint32_t owner;
    size_t capacity;
    size_t total;
    void* raw;
    struct ql_large_meta* prev;
    struct ql_large_meta* next;
} ql_large_meta;

struct ql_slab {
    uint32_t magic;
    uint16_t cls;
    uint16_t in_partial;
    uint32_t slot_count;
    uint32_t used;
    size_t used_bytes;
    size_t size;
    ql_obj_hdr* free_head;
    ql_slab* next_partial;
    ql_slab* all_prev;
    ql_slab* all_next;
    void* raw;
};

static ql_slab* ql_mem_slabs;
static ql_slab* ql_mem_partial[QL_MEM_CLASS_COUNT];
static ql_large_meta* ql_mem_larges;
static size_t ql_mem_block_size = QL_MEM_DEFAULT_BLOCK;
static size_t ql_mem_used_total;
static size_t ql_mem_live_total;
static size_t ql_mem_slab_bytes;
static long long ql_mem_alloc_total;
static long long ql_mem_free_total;
static long long ql_mem_reuse;
static long long ql_mem_new_blocks;
static long long ql_mem_slab_count;
static long long ql_mem_large_count;
static volatile int ql_mem_lock_word;

#if QL_MEM_DEBUG
static void ql_mem_assert_fail(int line, const char* expr) {
    fprintf(stderr, "ql_mem assertion failed at qthreads.c.txt:%d: %s\n", line, expr);
    fflush(stderr);
    abort();
}
#define QL_MEM_ASSERT(cond) do { if (!(cond)) ql_mem_assert_fail(__LINE__, #cond); } while (0)
#else
#define QL_MEM_ASSERT(cond) ((void)0)
#endif

/* Returns 1 when the lock was taken, 0 on the single-threaded fast path. The decision is returned
 * rather than re-read in unlock: the pool can enable the lock while a caller is running, and an
 * unlock that re-read the flag could release a lock this call never acquired. */
static int ql_mem_lock(void) {
#if defined(__GNUC__) || defined(__clang__)
    if (!__atomic_load_n(&ql_mem_multithreaded, __ATOMIC_ACQUIRE)) return 0;
    while (__atomic_exchange_n(&ql_mem_lock_word, 1, __ATOMIC_ACQUIRE)) {
#if defined(__x86_64__) || defined(__i386__)
        __builtin_ia32_pause();
#elif defined(__aarch64__) || defined(__arm__)
        __asm__ __volatile__("yield");
#endif
    }
    return 1;
#else
    return 0;
#endif
}

static void ql_mem_unlock(int locked) {
#if defined(__GNUC__) || defined(__clang__)
    if (!locked) return;
    __atomic_store_n(&ql_mem_lock_word, 0, __ATOMIC_RELEASE);
#else
    (void)locked;
#endif
}

static size_t ql_mem_slab_hdr(void) {
    return (sizeof(ql_slab) + QL_MEM_ALIGN - 1) & ~(size_t)(QL_MEM_ALIGN - 1);
}

static size_t ql_mem_stride(int cls) {
    size_t n = QL_MEM_OBJ_HDR + ql_mem_class_size[cls];
    return (n + QL_MEM_ALIGN - 1) & ~(size_t)(QL_MEM_ALIGN - 1);
}

/* A 16-byte-aligned host block. The host pointer is stored immediately before the aligned address;
 * allocating one alignment unit plus one pointer guarantees that slot is inside the allocation. */
static void* ql_mem_host_alloc(size_t size) {
    void* raw = malloc(size + QL_MEM_ALIGN + sizeof(void*));
    uintptr_t base;
    if (!raw) return 0;
    base = ((uintptr_t)raw + sizeof(void*) + QL_MEM_ALIGN - 1) & ~(uintptr_t)(QL_MEM_ALIGN - 1);
    ((void**)base)[-1] = raw;
    return (void*)base;
}

static void ql_mem_host_free(void* p) {
    if (p) free(((void**)p)[-1]);
}

static int ql_mem_class_of(size_t n) {
    int i;
    for (i = 0; i < QL_MEM_CLASS_COUNT; i++) {
        if (n <= ql_mem_class_size[i]) return i;
    }
    return -1;
}

static void ql_mem_obj_init(ql_obj_hdr* h, uint16_t kind, void* block, size_t requested) {
    h->magic = QL_MEM_OBJ_MAGIC;
    h->kind = kind;
    h->flags = QL_MEM_FLAG_LIVE;
    h->requested = requested;
    h->block = block;
    h->next_free = 0;
}

static size_t ql_mem_slab_size_for(int cls) {
    size_t min_size = ql_mem_slab_hdr() + ql_mem_stride(cls);
    size_t size = ql_mem_block_size;
    if (size < min_size) size = min_size;
    return (size + QL_MEM_ALIGN - 1) & ~(size_t)(QL_MEM_ALIGN - 1);
}

static void ql_mem_partial_push(int cls, ql_slab* s) {
    s->next_partial = ql_mem_partial[cls];
    ql_mem_partial[cls] = s;
    s->in_partial = 1;
}

static void ql_mem_partial_remove(int cls, ql_slab* s) {
    ql_slab** pp = &ql_mem_partial[cls];
    while (*pp && *pp != s) pp = &(*pp)->next_partial;
    if (*pp == s) *pp = s->next_partial;
    s->next_partial = 0;
    s->in_partial = 0;
}

static ql_slab* ql_mem_slab_new(int cls) {
    size_t stride = ql_mem_stride(cls);
    size_t size = ql_mem_slab_size_for(cls);
    uint32_t i;
    ql_slab* s;
    char* first;
    void* base = ql_mem_host_alloc(size);
    if (!base) return 0;
    s = (ql_slab*)base;
    s->magic = QL_MEM_SLAB_MAGIC;
    s->cls = (uint16_t)cls;
    s->in_partial = 0;
    s->slot_count = (uint32_t)((size - ql_mem_slab_hdr()) / stride);
    s->used = 0;
    s->used_bytes = 0;
    s->size = size;
    s->free_head = 0;
    s->next_partial = 0;
    s->all_prev = 0;
    s->all_next = ql_mem_slabs;
    if (ql_mem_slabs) ql_mem_slabs->all_prev = s;
    ql_mem_slabs = s;
    s->raw = base;
    if (s->slot_count == 0) {
        ql_mem_slabs = s->all_next;
        if (ql_mem_slabs) ql_mem_slabs->all_prev = 0;
        ql_mem_host_free(s);
        return 0;
    }
    first = (char*)base + ql_mem_slab_hdr();
    for (i = s->slot_count; i > 0; i--) {
        ql_obj_hdr* h = (ql_obj_hdr*)(first + (size_t)(i - 1) * stride);
        h->magic = QL_MEM_OBJ_MAGIC;
        h->kind = QL_MEM_KIND_SLOT;
        h->flags = 0;
        h->requested = 0;
        h->block = s;
        h->next_free = s->free_head;
        s->free_head = h;
    }
    ql_mem_slab_count++;
    ql_mem_slab_bytes += size;
    return s;
}

/* The caller holds the allocator lock. */
static void* ql_mem_alloc_locked(size_t n, int owner) {
    int cls;
    ql_slab* best;
    ql_obj_hdr* h;
    (void)owner;
    if (n == 0) n = 1;
    cls = ql_mem_class_of(n);
    if (cls >= 0) {
        size_t stride = ql_mem_stride(cls);
        ql_slab* s;
        best = 0;
        for (s = ql_mem_partial[cls]; s; s = s->next_partial) {
            QL_MEM_ASSERT(s->magic == QL_MEM_SLAB_MAGIC);
            QL_MEM_ASSERT(s->cls == (uint16_t)cls);
            QL_MEM_ASSERT(s->used <= s->slot_count);
            QL_MEM_ASSERT(s->size >= ql_mem_slab_hdr() + stride);
            if (s->free_head && (!best || s->used < best->used)) best = s;
        }
        if (!best) {
            best = ql_mem_slab_new(cls);
            if (!best) return 0;
            ql_mem_new_blocks++;
            if (best->slot_count > 1) ql_mem_partial_push(cls, best);
        } else {
            ql_mem_reuse++;
        }
        h = best->free_head;
        QL_MEM_ASSERT(h != 0);
        QL_MEM_ASSERT(h->magic == QL_MEM_OBJ_MAGIC);
        QL_MEM_ASSERT(h->kind == QL_MEM_KIND_SLOT);
        QL_MEM_ASSERT((h->flags & QL_MEM_FLAG_LIVE) == 0);
        QL_MEM_ASSERT(h->block == best);
        best->free_head = h->next_free;
        h->next_free = 0;
        h->flags = QL_MEM_FLAG_LIVE;
        h->requested = n;
        if (best->used == 0) ql_mem_live_total += best->size;
        best->used++;
        best->used_bytes += n;
        if (best->used == best->slot_count) {
            if (best->in_partial) ql_mem_partial_remove(cls, best);
        } else if (!best->in_partial) {
            ql_mem_partial_push(cls, best);
        }
        ql_mem_alloc_total++;
        ql_mem_used_total += n;
        QL_MEM_ASSERT(((uintptr_t)((char*)h + QL_MEM_OBJ_HDR) % QL_MEM_ALIGN) == 0);
        return (char*)h + QL_MEM_OBJ_HDR;
    }
    {
        size_t cap = (n + QL_MEM_ALIGN - 1) & ~(size_t)(QL_MEM_ALIGN - 1);
        size_t total;
        void* base;
        ql_large_meta* m;
        if (cap < n) return 0;
        total = sizeof(ql_large_meta) + QL_MEM_OBJ_HDR + cap;
        if (total < n) return 0;
        base = ql_mem_host_alloc(total);
        if (!base) return 0;
        m = (ql_large_meta*)base;
        h = (ql_obj_hdr*)((char*)base + sizeof(ql_large_meta));
        m->magic = QL_MEM_LARGE_MAGIC;
        m->owner = (uint32_t)owner;
        m->capacity = cap;
        m->total = total;
        m->raw = base;
        m->prev = 0;
        m->next = ql_mem_larges;
        if (ql_mem_larges) ql_mem_larges->prev = m;
        ql_mem_larges = m;
        ql_mem_large_count++;
        ql_mem_new_blocks++;
        ql_mem_obj_init(h, QL_MEM_KIND_LARGE, m, n);
        ql_mem_alloc_total++;
        ql_mem_live_total += total;
        ql_mem_used_total += n;
        QL_MEM_ASSERT(((uintptr_t)((char*)h + QL_MEM_OBJ_HDR) % QL_MEM_ALIGN) == 0);
        return (char*)h + QL_MEM_OBJ_HDR;
    }
}

static size_t ql_mem_obj_capacity(ql_obj_hdr* h) {
    if (h->kind == QL_MEM_KIND_SLOT) {
        ql_slab* s = (ql_slab*)h->block;
        return ql_mem_class_size[s->cls];
    }
    return ((ql_large_meta*)h->block)->capacity;
}

/* The caller holds the allocator lock. */
static void ql_mem_free_locked(void* p) {
    ql_obj_hdr* h;
    if (!p) return;
    h = (ql_obj_hdr*)((char*)p - QL_MEM_OBJ_HDR);
    QL_MEM_ASSERT(h->magic == QL_MEM_OBJ_MAGIC);
    QL_MEM_ASSERT((h->flags & QL_MEM_FLAG_LIVE) != 0);
    QL_MEM_ASSERT(((uintptr_t)p % QL_MEM_ALIGN) == 0);
    ql_mem_free_total++;
    if (h->kind == QL_MEM_KIND_SLOT) {
        ql_slab* s = (ql_slab*)h->block;
        QL_MEM_ASSERT(s->magic == QL_MEM_SLAB_MAGIC);
        QL_MEM_ASSERT(s->used > 0);
        QL_MEM_ASSERT(s->used_bytes >= h->requested);
        QL_MEM_ASSERT(h->requested <= ql_mem_class_size[s->cls]);
        h->flags = 0;
        h->next_free = s->free_head;
        s->free_head = h;
        s->used--;
        s->used_bytes -= h->requested;
        if (s->used == 0) ql_mem_live_total -= s->size;
        if (!s->in_partial) ql_mem_partial_push(s->cls, s);
        ql_mem_used_total -= h->requested;
        return;
    }
    QL_MEM_ASSERT(h->kind == QL_MEM_KIND_LARGE);
    {
        ql_large_meta* m = (ql_large_meta*)h->block;
        QL_MEM_ASSERT(m->magic == QL_MEM_LARGE_MAGIC);
        QL_MEM_ASSERT(m->capacity >= h->requested);
        QL_MEM_ASSERT(m->total >= sizeof(ql_large_meta) + QL_MEM_OBJ_HDR);
        if (m->prev) m->prev->next = m->next;
        else ql_mem_larges = m->next;
        if (m->next) m->next->prev = m->prev;
        ql_mem_large_count--;
        ql_mem_used_total -= h->requested;
        ql_mem_live_total -= m->total;
        ql_mem_host_free(m);
    }
}

void ql_free(void* ptr);

void* ql_alloc(size_t size, int owner) {
    void* p;
    int locked = ql_mem_lock();
    p = ql_mem_alloc_locked(size, owner);
    ql_mem_unlock(locked);
    return p;
}

void* ql_calloc(size_t count, size_t size, int owner) {
    size_t n;
    void* p;
    if (size != 0 && count > (size_t)-1 / size) return 0;
    n = count * size;
    int locked = ql_mem_lock();
    p = ql_mem_alloc_locked(n, owner);
    if (p) memset(p, 0, n);
    ql_mem_unlock(locked);
    return p;
}

void* ql_realloc(void* ptr, size_t size) {
    ql_obj_hdr* h;
    size_t old_requested;
    size_t capacity;
    void* fresh;
    if (!ptr) return ql_alloc(size, 0);
    if (size == 0) {
        ql_free(ptr);
        return 0;
    }
    int locked = ql_mem_lock();
    h = (ql_obj_hdr*)((char*)ptr - QL_MEM_OBJ_HDR);
    QL_MEM_ASSERT(h->magic == QL_MEM_OBJ_MAGIC);
    QL_MEM_ASSERT((h->flags & QL_MEM_FLAG_LIVE) != 0);
    old_requested = h->requested;
    capacity = ql_mem_obj_capacity(h);
    if (size <= capacity) {
        if (h->kind == QL_MEM_KIND_SLOT) {
            ql_slab* s = (ql_slab*)h->block;
            if (size >= old_requested) s->used_bytes += size - old_requested;
            else s->used_bytes -= old_requested - size;
        }
        if (size >= old_requested) ql_mem_used_total += size - old_requested;
        else ql_mem_used_total -= old_requested - size;
        h->requested = size;
        ql_mem_unlock(locked);
        return ptr;
    }
    /* The list growth paths ask for 2*tail elements, so once tail passes half the capacity the
     * request grows by one element at a time. A realloc implementation that always allocates exactly
     * the request would copy the whole buffer on nearly every append (O(n^2)); the host realloc gets
     * away with it because it can often extend in place. The counted-slab large path cannot grow in
     * place, so it keeps the doubled capacity it allocated and satisfies subsequent grew-in-place
     * requests from it. This is ordinary realloc slack (the returned block may be larger than asked),
     * only the bookkeeping reports the requested size. */
    {
        size_t grow = size;
        ql_obj_hdr* nh;
        if (size > ql_mem_class_size[QL_MEM_CLASS_COUNT - 1] && capacity <= (size_t)-1 / 2) {
            size_t doubled = capacity * 2;
            if (doubled > grow) grow = doubled;
        }
        fresh = ql_mem_alloc_locked(grow, 0);
        if (!fresh) {
            ql_mem_unlock(locked);
            return 0;
        }
        nh = (ql_obj_hdr*)((char*)fresh - QL_MEM_OBJ_HDR);
        if (grow != size) {
            QL_MEM_ASSERT(nh->kind == QL_MEM_KIND_LARGE);
            QL_MEM_ASSERT(nh->requested == grow);
            ql_mem_used_total -= grow - size;
            nh->requested = size;
        }
    }
    memcpy(fresh, ptr, old_requested < size ? old_requested : size);
    ql_mem_free_locked(ptr);
    ql_mem_unlock(locked);
    return fresh;
}

void ql_free(void* ptr) {
    if (!ptr) return;
    int locked = ql_mem_lock();
    ql_mem_free_locked(ptr);
    ql_mem_unlock(locked);
}

int ql_mem_set_block(int n) {
    if (n < 1 || n > QL_MEM_MAX_BLOCK) return 0;
    int locked = ql_mem_lock();
    ql_mem_block_size = (size_t)n;
    ql_mem_unlock(locked);
    return 1;
}

void ql_mem_compact(void) {
    ql_slab* s;
    ql_slab* next;
    int cls;
    int locked = ql_mem_lock();
    s = ql_mem_slabs;
    while (s) {
        next = s->all_next;
        if (s->used == 0) {
            if (s->all_prev) s->all_prev->all_next = s->all_next;
            else ql_mem_slabs = s->all_next;
            if (s->all_next) s->all_next->all_prev = s->all_prev;
            ql_mem_slab_count--;
            ql_mem_slab_bytes -= s->size;
            ql_mem_host_free(s);
        }
        s = next;
    }
    for (cls = 0; cls < QL_MEM_CLASS_COUNT; cls++) ql_mem_partial[cls] = 0;
    for (s = ql_mem_slabs; s; s = s->all_next) {
        s->next_partial = 0;
        s->in_partial = 0;
        if (s->free_head) ql_mem_partial_push(s->cls, s);
    }
    ql_mem_unlock(locked);
}

double ql_mem_fragmentation(void) {
    double r;
    int locked = ql_mem_lock();
    if (ql_mem_live_total == 0) r = 0.0;
    else r = (double)(ql_mem_live_total - ql_mem_used_total) / (double)ql_mem_live_total;
    ql_mem_unlock(locked);
    return r;
}

double ql_mem_occupancy(void) {
    double r;
    int locked = ql_mem_lock();
    if (ql_mem_live_total == 0) r = 0.0;
    else r = (double)ql_mem_used_total / (double)ql_mem_live_total;
    ql_mem_unlock(locked);
    return r;
}

long long ql_mem_block_count(void) {
    long long n;
    int locked = ql_mem_lock();
    n = ql_mem_slab_count + ql_mem_large_count;
    ql_mem_unlock(locked);
    return n;
}

long long ql_mem_live_bytes(void) {
    long long n;
    int locked = ql_mem_lock();
    n = (long long)ql_mem_live_total;
    ql_mem_unlock(locked);
    return n;
}

long long ql_mem_used_bytes(void) {
    long long n;
    int locked = ql_mem_lock();
    n = (long long)ql_mem_used_total;
    ql_mem_unlock(locked);
    return n;
}

long long ql_mem_alloc_calls(void) {
    long long n;
    int locked = ql_mem_lock();
    n = ql_mem_alloc_total;
    ql_mem_unlock(locked);
    return n;
}

long long ql_mem_free_calls(void) {
    long long n;
    int locked = ql_mem_lock();
    n = ql_mem_free_total;
    ql_mem_unlock(locked);
    return n;
}

void ql_mem_stats(long long* reuse, long long* new_blocks, long long* block_count) {
    int locked = ql_mem_lock();
    if (reuse) *reuse = ql_mem_reuse;
    if (new_blocks) *new_blocks = ql_mem_new_blocks;
    if (block_count) *block_count = ql_mem_slab_count + ql_mem_large_count;
    ql_mem_unlock(locked);
}

static int ql_mem_check_one(const char* what) {
    fprintf(stderr, "ql_mem_check: %s\n", what);
    return 1;
}

/* Full invariant walk. It is O(all slots + all large blocks); the debug assertions on the hot path
 * are the cheap subset, and the tests call this after a batch of operations. Returns 0 on success. */
static int ql_mem_check_locked(void) {
    int errors = 0;
    long long slab_count = 0;
    size_t slab_bytes = 0;
    size_t live_bytes = 0;
    size_t used_bytes = 0;
    long long large_count = 0;
    ql_slab* s;
    ql_large_meta* m;
    if (ql_mem_alloc_total != ql_mem_reuse + ql_mem_new_blocks) errors += ql_mem_check_one("alloc_calls != reuse + new_blocks");
    for (s = ql_mem_slabs; s; s = s->all_next) {
        size_t stride;
        uint32_t i;
        uint32_t live = 0;
        uint32_t free_slots = 0;
        ql_obj_hdr* h;
        ql_obj_hdr* f;
        if (s->magic != QL_MEM_SLAB_MAGIC) { errors += ql_mem_check_one("slab magic"); continue; }
        if (s->cls >= QL_MEM_CLASS_COUNT) { errors += ql_mem_check_one("slab class"); continue; }
        if (s->all_next && s->all_next->all_prev != s) errors += ql_mem_check_one("slab all_next->all_prev");
        if (s->used > s->slot_count) errors += ql_mem_check_one("slab used > slot_count");
        if (s->used_bytes > s->size) errors += ql_mem_check_one("slab used_bytes > size");
        if (s->size < ql_mem_slab_hdr() + ql_mem_stride(s->cls)) errors += ql_mem_check_one("slab too small for one slot");
        stride = ql_mem_stride(s->cls);
        if (s->slot_count != (uint32_t)((s->size - ql_mem_slab_hdr()) / stride)) errors += ql_mem_check_one("slab slot_count");
        for (i = 0; i < s->slot_count; i++) {
            h = (ql_obj_hdr*)((char*)s + ql_mem_slab_hdr() + (size_t)i * stride);
            if (h->magic != QL_MEM_OBJ_MAGIC) { errors += ql_mem_check_one("slot magic"); continue; }
            if (h->kind != QL_MEM_KIND_SLOT) errors += ql_mem_check_one("slot kind");
            if (h->block != s) errors += ql_mem_check_one("slot block");
            if (h->flags & QL_MEM_FLAG_LIVE) {
                live++;
                if (h->requested > ql_mem_class_size[s->cls]) errors += ql_mem_check_one("slot requested > class");
                used_bytes += h->requested;
            }
            if (((uintptr_t)((char*)h + QL_MEM_OBJ_HDR) % QL_MEM_ALIGN) != 0) errors += ql_mem_check_one("slot alignment");
        }
        for (f = s->free_head; f; f = f->next_free) {
            free_slots++;
            if (free_slots > s->slot_count) { errors += ql_mem_check_one("free list cycle"); break; }
            if (f->magic != QL_MEM_OBJ_MAGIC) { errors += ql_mem_check_one("free slot magic"); break; }
            if (f->kind != QL_MEM_KIND_SLOT) { errors += ql_mem_check_one("free slot kind"); break; }
            if (f->block != s) { errors += ql_mem_check_one("free slot block"); break; }
            if (f->flags & QL_MEM_FLAG_LIVE) { errors += ql_mem_check_one("free slot live"); break; }
        }
        if (live + free_slots != s->slot_count) errors += ql_mem_check_one("slab live + free != slot_count");
        if (live != s->used) errors += ql_mem_check_one("slab used != live slots");
        {
            size_t slab_live_bytes = 0;
            for (i = 0; i < s->slot_count; i++) {
                h = (ql_obj_hdr*)((char*)s + ql_mem_slab_hdr() + (size_t)i * stride);
                if (h->flags & QL_MEM_FLAG_LIVE) slab_live_bytes += h->requested;
            }
            if (slab_live_bytes != s->used_bytes) errors += ql_mem_check_one("slab used_bytes mismatch");
        }
        if (s->in_partial) {
            ql_slab* p = ql_mem_partial[s->cls];
            while (p && p != s) p = p->next_partial;
            if (p != s) errors += ql_mem_check_one("in_partial slab missing from partial list");
        }
        if (s->free_head && !s->in_partial) errors += ql_mem_check_one("free slot on a slab not in partial");
        if (!s->free_head && s->in_partial) errors += ql_mem_check_one("full slab still in partial");
        if (s->used > 0) live_bytes += s->size;
        slab_count++;
        slab_bytes += s->size;
    }
    for (m = ql_mem_larges; m; m = m->next) {
        ql_obj_hdr* h = (ql_obj_hdr*)((char*)m + sizeof(ql_large_meta));
        if (m->magic != QL_MEM_LARGE_MAGIC) { errors += ql_mem_check_one("large magic"); continue; }
        if (m->next && m->next->prev != m) errors += ql_mem_check_one("large next->prev");
        if (h->magic != QL_MEM_OBJ_MAGIC) { errors += ql_mem_check_one("large object magic"); continue; }
        if (h->kind != QL_MEM_KIND_LARGE) errors += ql_mem_check_one("large kind");
        if (h->block != m) errors += ql_mem_check_one("large block");
        if (!(h->flags & QL_MEM_FLAG_LIVE)) errors += ql_mem_check_one("large not live");
        if (h->requested > m->capacity) errors += ql_mem_check_one("large requested > capacity");
        if (m->total != sizeof(ql_large_meta) + QL_MEM_OBJ_HDR + m->capacity) errors += ql_mem_check_one("large total");
        if (((uintptr_t)((char*)h + QL_MEM_OBJ_HDR) % QL_MEM_ALIGN) != 0) errors += ql_mem_check_one("large alignment");
        live_bytes += m->total;
        used_bytes += h->requested;
        large_count++;
    }
    if (slab_count != ql_mem_slab_count) errors += ql_mem_check_one("slab_count counter");
    if (slab_bytes != ql_mem_slab_bytes) errors += ql_mem_check_one("slab_bytes counter");
    if (large_count != ql_mem_large_count) errors += ql_mem_check_one("large_count counter");
    if (live_bytes != ql_mem_live_total) errors += ql_mem_check_one("live_bytes counter");
    if (used_bytes != ql_mem_used_total) errors += ql_mem_check_one("used_bytes counter");
    return errors;
}

int ql_mem_check(void) {
    int errors;
    int locked = ql_mem_lock();
    if (sizeof(ql_obj_hdr) != QL_MEM_OBJ_HDR) errors = ql_mem_check_one("object header size");
    else errors = ql_mem_check_locked();
    ql_mem_unlock(locked);
    return errors;
}

/* ================= Messages =================
 * The strings this runtime prints come from the generated C catalog (compiler/internal/rtcat/msg_c.h,
 * from the authored tables under internal/i18n/source through cmd/qki18n): the same rows, languages and
 * verbs as the interpreter's catalog, so a compiled program and `quark` say the same thing in the same
 * language. qkc prepends the header, so nothing here is compiled without it.
 *
 * The language is the process environment, resolved when a message is first rendered. The order — and
 * the prefix match — is the Go side's Detect(): QK_LANG, then LC_ALL / LC_MESSAGES / LANG, then the
 * source language. Reading the environment at render time rather than at start-up is what keeps
 * `QK_LANG=en prog` and `QK_LANG=zh prog` the same binary.
 */
static int qk_lang_block(void) {
    static int cached = -1;
    const char* names[4];
    if (cached >= 0) return cached;
    names[0] = getenv("QK_LANG");
    names[1] = getenv("LC_ALL");
    names[2] = getenv("LC_MESSAGES");
    names[3] = getenv("LANG");
    for (int i = 0; i < 4; i++) {
        const char* v = names[i];
        if (!v) continue;
        while (*v == ' ' || *v == '\t') v++;
        if ((v[0] == 'z' || v[0] == 'Z') && (v[1] == 'h' || v[1] == 'H')) { cached = 1; return cached; }
        if ((v[0] == 'e' || v[0] == 'E') && (v[1] == 'n' || v[1] == 'N')) { cached = 0; return cached; }
    }
    cached = 0;
    return cached;
}

/* qk_msg_text renders one message of the catalog into buf. A catalog id is generated code, so an id
 * the header does not carry is a build mistake, not a user-facing condition; rendering the id keeps
 * the failure visible instead of printing an empty message. */
static const char* qk_msg_text(int lang, const char* id, char* buf, size_t n) {
    const char* s = qk_msg(lang, id);
    size_t len;
    if (!s) {
        snprintf(buf, n, "missing message: %s", id);
        return buf;
    }
    len = strlen(s);
    if (len >= n) len = n - 1;
    memcpy(buf, s, len);
    buf[len] = 0;
    return buf;
}

/* qk_quote mirrors the interpreter's %q for the strings that reach a parse error (Go's strconv.Quote
 * over printable ASCII and the UTF-8 bytes): quotes and backslashes are escaped, the C escapes Go
 * spells with a letter are the same letters, and any other control byte is a two-digit hex escape. The
 * two sides therefore quote the same value the same way, which is what keeps one message one message. */
static void qk_quote(const char* s, char* out, size_t n) {
    size_t o = 0;
    if (n) out[o++] = '"';
    for (const unsigned char* p = (const unsigned char*)s; *p && o + 8 < n; p++) {
        unsigned char c = *p;
        switch (c) {
        case '"': out[o++] = '\\'; out[o++] = '"'; break;
        case '\\': out[o++] = '\\'; out[o++] = '\\'; break;
        case '\a': out[o++] = '\\'; out[o++] = 'a'; break;
        case '\b': out[o++] = '\\'; out[o++] = 'b'; break;
        case '\f': out[o++] = '\\'; out[o++] = 'f'; break;
        case '\n': out[o++] = '\\'; out[o++] = 'n'; break;
        case '\r': out[o++] = '\\'; out[o++] = 'r'; break;
        case '\t': out[o++] = '\\'; out[o++] = 't'; break;
        case '\v': out[o++] = '\\'; out[o++] = 'v'; break;
        default:
            if (c < 0x20 || c == 0x7f) {
                static const char hex[] = "0123456789abcdef";
                out[o++] = '\\'; out[o++] = 'x'; out[o++] = hex[c >> 4]; out[o++] = hex[c & 15];
            } else {
                out[o++] = (char)c;
            }
            break;
        }
    }
    if (o < n) out[o++] = '"';
    if (o < n) out[o] = 0;
    else if (n) out[n - 1] = 0;
}

/* ================= String built-in methods (same semantics as the interpreter internal/lang/eval.go) =================
 * rune (UTF-8) semantics: size/charAt/substring are per rune; contains/startsWith/endsWith/indexOf
 * are per byte (same semantics as the Go strings package). Out-of-range / parse failure → the same runtime error wording as the interpreter, exiting 1. */
static void ql_err_at(const char* msg, int line) {
    fprintf(stderr, "error: %s at line %d\n", msg, line);
    fflush(stderr);
    exit(1);
}
static int ql_u8seq(unsigned char c) {
    if (c < 0x80) return 1;
    if ((c & 0xE0) == 0xC0) return 2;
    if ((c & 0xF0) == 0xE0) return 3;
    if ((c & 0xF8) == 0xF0) return 4;
    return 1;
}
int ql_str_size(const char* s) {
    int n = 0;
    for (const unsigned char* p = (const unsigned char*)s; *p; ) { p += ql_u8seq(*p); n++; }
    return n;
}
/* ql_u8_off: byte offset of the i-th rune (i == rune count → strlen; i beyond → -1) */
static int ql_u8_off(const char* s, int i) {
    int b = 0;
    for (int k = 0; k < i; k++) {
        if (!s[b]) return -1;
        b += ql_u8seq((unsigned char)s[b]);
    }
    return b;
}
int ql_str_contains(const char* s, const char* sub) { return strstr(s, sub) != 0; }
int ql_str_startswith(const char* s, const char* p) { size_t lp = strlen(p); return strncmp(s, p, lp) == 0; }
int ql_str_endswith(const char* s, const char* p) {
    size_t ls = strlen(s), lp = strlen(p);
    return lp <= ls && memcmp(s + ls - lp, p, lp) == 0;
}
int ql_str_indexof(const char* s, const char* sub) {
    const char* p = strstr(s, sub);
    return p ? (int)(p - s) : -1;
}
char* ql_str_sub(const char* s, int start, int end, int line) {
    int n = ql_str_size(s);
    if (start < 0 || end < start || end > n) {
        char msg[192], text[192];
        qk_msg_text(qk_lang_block(), "runtime.stringindexoutofboundserror_substring_d_d", text, sizeof text);
        snprintf(msg, sizeof msg, text, start, end, n);
        ql_err_at(msg, line);
    }
    int b0 = ql_u8_off(s, start), b1 = ql_u8_off(s, end);
    int len = b1 - b0;
    char* r = malloc(len + 1);
    memcpy(r, s + b0, len); r[len] = 0;
    return r;
}
char* ql_str_charat(const char* s, int i, int line) {
    int n = ql_str_size(s);
    int b = ql_u8_off(s, i);
    if (i < 0 || b < 0 || s[b] == 0) {
        char msg[192], text[192];
        qk_msg_text(qk_lang_block(), "runtime.stringindexoutofboundserror_charat_d_out", text, sizeof text);
        snprintf(msg, sizeof msg, text, i, n);
        ql_err_at(msg, line);
    }
    int sz = ql_u8seq((unsigned char)s[b]);
    char* r = malloc(sz + 1);
    memcpy(r, s + b, sz); r[sz] = 0;
    return r;
}
/* The whitespace set of Go unicode.IsSpace (including the Zs category and U+0085) */
static int ql_space_at(const char* s, int* sz) {
    unsigned char c = (unsigned char)s[0];
    if (c == ' ' || c == '\t' || c == '\n' || c == '\v' || c == '\f' || c == '\r') { *sz = 1; return 1; }
    if (c == 0xC2 && (unsigned char)s[1] == 0x85) { *sz = 2; return 1; } /* U+0085 */
    if (c == 0xC2 && (unsigned char)s[1] == 0xA0) { *sz = 2; return 1; } /* U+00A0 */
    if (c == 0xE1 && (unsigned char)s[1] == 0x9A && (unsigned char)s[2] == 0x80) { *sz = 3; return 1; } /* U+1680 */
    if (c == 0xE2 && (unsigned char)s[1] == 0x80) {
        unsigned char d = (unsigned char)s[2];
        if (d >= 0x80 && d <= 0x8A) { *sz = 3; return 1; } /* U+2000..U+200A */
        if (d == 0xA8 || d == 0xA9) { *sz = 3; return 1; } /* U+2028/2029 */
        if (d == 0xAF) { *sz = 3; return 1; }              /* U+202F */
    }
    if (c == 0xE2 && (unsigned char)s[1] == 0x81 && (unsigned char)s[2] == 0x9F) { *sz = 3; return 1; } /* U+205F */
    if (c == 0xE3 && (unsigned char)s[1] == 0x80 && (unsigned char)s[2] == 0x80) { *sz = 3; return 1; } /* U+3000 */
    return 0;
}
char* ql_str_trim(const char* s, int mode) { /* 0=both 1=left 2=right */
    int len = (int)strlen(s);
    int b = 0, e = len;
    if (mode != 2) {
        while (b < e) { int sz = 0; if (!ql_space_at(s + b, &sz) || b + sz > e) break; b += sz; }
    }
    if (mode != 1) {
        while (e > b) {
            int k = e - 1;
            while (k > b && ((unsigned char)s[k] & 0xC0) == 0x80) k--;
            int sz = 0;
            if (!ql_space_at(s + k, &sz)) break;
            e = k;
        }
    }
    int n = e - b;
    char* r = malloc(n + 1);
    memcpy(r, s + b, n); r[n] = 0;
    return r;
}
/* Case mapping for ASCII + Latin-1 + the basic Greek/Cyrillic ranges (other scripts are left as-is) */
static int ql_lower_rune(int cp) {
    if (cp >= 'A' && cp <= 'Z') return cp + 32;
    if (cp >= 0xC0 && cp <= 0xDE && cp != 0xD7) return cp + 32;
    if (cp >= 0x391 && cp <= 0x3A9) return cp + 32;
    if (cp >= 0x410 && cp <= 0x42F) return cp + 32;
    if (cp >= 0x400 && cp <= 0x40F) return cp + 80;
    return cp;
}
static int ql_upper_rune(int cp) {
    if (cp >= 'a' && cp <= 'z') return cp - 32;
    if (cp >= 0xE0 && cp <= 0xFE && cp != 0xF7) return cp - 32;
    if (cp >= 0x3B1 && cp <= 0x3C9) return cp - 32;
    if (cp >= 0x430 && cp <= 0x44F) return cp - 32;
    if (cp >= 0x450 && cp <= 0x45F) return cp - 80;
    return cp;
}
static int ql_u8dec(const char* s, int* cp) {
    unsigned char c = (unsigned char)s[0];
    if (c < 0x80) { *cp = c; return 1; }
    if ((c & 0xE0) == 0xC0) { *cp = ((c & 0x1F) << 6) | ((unsigned char)s[1] & 0x3F); return 2; }
    if ((c & 0xF0) == 0xE0) { *cp = ((c & 0x0F) << 12) | (((unsigned char)s[1] & 0x3F) << 6) | ((unsigned char)s[2] & 0x3F); return 3; }
    if ((c & 0xF8) == 0xF0) { *cp = ((c & 0x07) << 18) | (((unsigned char)s[1] & 0x3F) << 12) | (((unsigned char)s[2] & 0x3F) << 6) | ((unsigned char)s[3] & 0x3F); return 4; }
    *cp = c; return 1;
}
static int ql_u8enc(int cp, char* out) {
    if (cp < 0x80) { out[0] = (char)cp; return 1; }
    if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
    if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
    out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F)); out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
}
static char* ql_str_mapcase(const char* s, int upper) {
    size_t l = strlen(s);
    char* r = malloc(l * 4 + 4);
    size_t o = 0;
    for (size_t i = 0; i < l; ) {
        int cp = 0, sz = ql_u8dec(s + i, &cp);
        cp = upper ? ql_upper_rune(cp) : ql_lower_rune(cp);
        char tmp[4];
        int n = ql_u8enc(cp, tmp);
        memcpy(r + o, tmp, n); o += n; i += sz;
    }
    r[o] = 0;
    return r;
}
char* ql_str_lower(const char* s) { return ql_str_mapcase(s, 0); }
char* ql_str_upper(const char* s) { return ql_str_mapcase(s, 1); }
char* ql_str_replace(const char* s, const char* old, const char* nw) {
    size_t lo = strlen(old), ln = strlen(nw);
    if (lo == 0) {
        size_t l = strlen(s), o = 0;
        char* r = malloc((l + 1) * (ln + 4) + 1);
        for (size_t i = 0; i < l; ) {
            memcpy(r + o, nw, ln); o += ln;
            int sz = ql_u8seq((unsigned char)s[i]);
            memcpy(r + o, s + i, sz); o += sz; i += sz;
        }
        memcpy(r + o, nw, ln); o += ln;
        r[o] = 0;
        return r;
    }
    size_t l = strlen(s), cap = l + 1, o = 0;
    char* r = malloc(cap);
    const char* p = s;
    for (;;) {
        const char* hit = strstr(p, old);
        size_t seg = hit ? (size_t)(hit - p) : strlen(p);
        if (o + seg + ln + 1 > cap) { cap = (o + seg + ln + 1) * 2; r = realloc(r, cap); }
        memcpy(r + o, p, seg); o += seg;
        if (!hit) break;
        memcpy(r + o, nw, ln); o += ln;
        p = hit + lo;
    }
    r[o] = 0;
    return r;
}
/* ql_parse_err: the value did not parse. The two callers pass the catalog id of their row; the row is
 * a sentence with one %q in it, so this renders the quoted value into it. Go spells "the value,
 * quoted" with %q and C's printf has no such verb, so the verb is rewritten to %s here — the quoting
 * is the same on both sides (qk_quote mirrors strconv.Quote for the strings that reach a parse error),
 * which is what keeps one message one message instead of two rows that could drift apart. */
static void ql_parse_err(const char* s, const char* id, int line) {
    char quoted[256], text[192], fmt[192], msg[256];
    size_t k = 0;
    qk_quote(s, quoted, sizeof quoted);
    qk_msg_text(qk_lang_block(), id, text, sizeof text);
    for (size_t i = 0; text[i] && k + 3 < sizeof fmt; i++) {
        if (text[i] == '%' && text[i + 1] == 'q') {
            fmt[k++] = '%'; fmt[k++] = 's'; i++;
        } else if (text[i] == '%' && text[i + 1] == '%') {
            fmt[k++] = '%'; fmt[k++] = '%'; i++;
        } else {
            fmt[k++] = text[i];
        }
    }
    fmt[k] = 0;
    snprintf(msg, sizeof msg, fmt, quoted);
    ql_err_at(msg, line);
}
int ql_str_toint(const char* s, int line) {
    const char* p = s;
    while (*p == ' ' || *p == '\t' || *p == '\n' || *p == '\v' || *p == '\f' || *p == '\r') p++;
    const char* q = p + strlen(p);
    while (q > p && (q[-1] == ' ' || q[-1] == '\t' || q[-1] == '\n' || q[-1] == '\v' || q[-1] == '\f' || q[-1] == '\r')) q--;
    char buf[64];
    size_t n = (size_t)(q - p);
    if (n == 0 || n >= sizeof buf) ql_parse_err(s, "runtime.parseerror_q_valid_integer", line);
    memcpy(buf, p, n); buf[n] = 0;
    char* end = 0;
    long v = strtol(buf, &end, 10);
    if (end == buf || *end != 0 || v < -2147483648L || v > 2147483647L) ql_parse_err(s, "runtime.parseerror_q_valid_integer", line);
    return (int)v;
}
double ql_str_tofloat(const char* s, int line) {
    const char* p = s;
    while (*p == ' ' || *p == '\t' || *p == '\n' || *p == '\v' || *p == '\f' || *p == '\r') p++;
    const char* q = p + strlen(p);
    while (q > p && (q[-1] == ' ' || q[-1] == '\t' || q[-1] == '\n' || q[-1] == '\v' || q[-1] == '\f' || q[-1] == '\r')) q--;
    char buf[128];
    size_t n = (size_t)(q - p);
    if (n == 0 || n >= sizeof buf) ql_parse_err(s, "runtime.parseerror_q_valid_float", line);
    memcpy(buf, p, n); buf[n] = 0;
    char* end = 0;
    double v = strtod(buf, &end);
    if (end == buf || *end != 0) ql_parse_err(s, "runtime.parseerror_q_valid_float", line);
    return v;
}

/* ================= List built-in method helpers (int elements; same semantics as the interpreter value.go) ================= */
static int ql_cmp_int(const void* a, const void* b) {
    int x = *(const int*)a, y = *(const int*)b;
    return x < y ? -1 : (x > y ? 1 : 0);
}
void ql_list_int_sort(int* buf, int head, int tail) {
    if (tail - head > 1) qsort(buf + head, tail - head, sizeof(int), ql_cmp_int);
}
/* List.toString(): [e0, e1, ...] (only the visible range head..tail) */
char* ql_list_int_str(const int* buf, int head, int tail) {
    size_t cap = 16, o = 0;
    char* r = malloc(cap);
    r[o++] = '[';
    for (int i = head; i < tail; i++) {
        char nb[16];
        int n = snprintf(nb, sizeof nb, "%d", buf[i]);
        if (i > head) {
            if (o + 2 > cap) { cap *= 2; r = realloc(r, cap); }
            r[o++] = ','; r[o++] = ' ';
        }
        if (o + (size_t)n + 2 > cap) { cap = (o + n + 2) * 2; r = realloc(r, cap); }
        memcpy(r + o, nb, n); o += (size_t)n;
    }
    if (o + 2 > cap) { cap += 2; r = realloc(r, cap); }
    r[o++] = ']'; r[o] = 0;
    return r;
}

/* ================= Signature memorize runtime (@mb() memoization) =================
 * key = the argument list (int array), value = the wrapped function int return value;
 * semantics match the interpreter MemorizeBuffer: a hit returns directly, a miss executes and records. */
typedef struct ql_memo_ent { int n; int* keys; int val; struct ql_memo_ent* next; } ql_memo_ent;
typedef struct { ql_memo_ent** b; int cap; int len; } ql_memo;
static unsigned ql_memo_hash(const int* keys, int n) {
    unsigned h = 2166136261u; /* FNV-1a */
    for (int i = 0; i < n; i++) { h ^= (unsigned)keys[i]; h *= 16777619u; }
    return h;
}
void* ql_memo_new(void) {
    ql_memo* m = (ql_memo*)calloc(1, sizeof(ql_memo));
    m->cap = 64; m->len = 0;
    m->b = (ql_memo_ent**)calloc((size_t)m->cap, sizeof(ql_memo_ent*));
    return m;
}
static ql_memo_ent* ql_memo_find(ql_memo* m, const int* keys, int n) {
    ql_memo_ent* e = m->b[ql_memo_hash(keys, n) % (unsigned)m->cap];
    while (e) {
        if (e->n == n) {
            int i = 0;
            while (i < n && e->keys[i] == keys[i]) i++;
            if (i == n) return e;
        }
        e = e->next;
    }
    return 0;
}
int ql_memo_get(void* mp, int n, int* keys, int* found) {
    ql_memo* m = (ql_memo*)mp;
    ql_memo_ent* e = ql_memo_find(m, keys, n);
    *found = e ? 1 : 0;
    return e ? e->val : 0;
}
void ql_memo_put(void* mp, int n, int* keys, int val) {
    ql_memo* m = (ql_memo*)mp;
    if (ql_memo_find(m, keys, n)) return; /* already recorded: keep the first result (the interpreter Put overwrites, but same key means same value) */
    ql_memo_ent* e = (ql_memo_ent*)calloc(1, sizeof(ql_memo_ent));
    e->n = n; e->val = val;
    e->keys = (int*)malloc(sizeof(int) * (size_t)(n > 0 ? n : 1));
    for (int i = 0; i < n; i++) e->keys[i] = keys[i];
    unsigned h = ql_memo_hash(keys, n) % (unsigned)m->cap;
    e->next = m->b[h]; m->b[h] = e; m->len++;
}

/* ================= interface{} (tAny) runtime: RTTI descriptor dispatch =================
 * the descriptor layout must match the IR %RT = type { i8* (i8*)*, i8*, i32 }:
 *   { char* (*str)(void*); const char* name; int kind; }
 * kind: 0=int 1=float 2=bool 3=String 4=List<int> 5=struct 6=null */
typedef struct ql_rt { char* (*str)(void*); void* (*copy)(void*); const char* name; int kind; } ql_rt;
char* ql_int_to_str(int v);      /* IR helper generated by qkc (reuses the same formatting) */
char* ql_float_to_str(double v);
char* ql_long_to_str(long long v) {
    char* b = (char*)malloc(24);
    snprintf(b, 24, "%lld", v);
    return b;
}
char* ql_any_str_int(void* d) { return ql_int_to_str(*(int*)d); }
char* ql_any_str_float(void* d) { return ql_float_to_str(*(double*)d); }
char* ql_any_str_bool(void* d) { return (char*)(*(int*)d ? "true" : "false"); }
char* ql_any_str_string(void* d) { return *(char**)d; }
char* ql_any_str_list(void* d);
char* ql_any_str(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind == 6 || !d) return (char*)"nil";
    return r->str(d);
}
static void ql_any_bad(ql_rt* r, const char* want) {
    fprintf(stderr, "error: TypeError: cannot use interface{} value of type %s as %s\n",
            r ? r->name : "nil", want);
    exit(1);
}
int ql_any_int(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 0) ql_any_bad(r, "int");
    return *(int*)d;
}
double ql_any_float(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 1) ql_any_bad(r, "float");
    return *(double*)d;
}
int ql_any_bool(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 2) ql_any_bad(r, "bool");
    return *(int*)d;
}
char* ql_any_string(void* d, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    if (!r || r->kind != 3) ql_any_bad(r, "String");
    return *(char**)d;
}
/* Equality semantics aligned with the interpreter equalValues: int/float compare numerically across types; bool/String by value;
 * List/struct by reference (pointer identity); nil == nil. */
int ql_any_eq(void* d1, void* rt1, void* d2, void* rt2) {
    ql_rt* a = (ql_rt*)rt1;
    ql_rt* b = (ql_rt*)rt2;
    int k1 = a ? a->kind : 6, k2 = b ? b->kind : 6;
    if (k1 == 6 || k2 == 6) return k1 == k2;
    if ((k1 == 0 || k1 == 1) && (k2 == 0 || k2 == 1)) {
        double x = (k1 == 0) ? (double)*(int*)d1 : *(double*)d1;
        double y = (k2 == 0) ? (double)*(int*)d2 : *(double*)d2;
        return x == y;
    }
    if (k1 != k2) return 0;
    switch (k1) {
    case 2: return (*(int*)d1 != 0) == (*(int*)d2 != 0);
    case 3: return strcmp(*(char**)d1, *(char**)d2) == 0;
    case 4: case 5: return d1 == d2;
    }
    return 0;
}
/* ---- deep copy of interface{} values (per RTTI; reused by HashTable.Put and copyd) ----
 * semantics aligned with the interpreter deepCopy: List copies the buffer, struct is not copied (pointer shared),
 * String/scalars are snapshotted by value (heap cell copied). */
void* ql_any_copy_int(void* d) { int* p = (int*)malloc(4); *p = *(int*)d; return p; }
void* ql_any_copy_float(void* d) { double* p = (double*)malloc(8); *p = *(double*)d; return p; }
void* ql_any_copy_bool(void* d) { int* p = (int*)malloc(4); *p = *(int*)d; return p; }
void* ql_any_copy_string(void* d) { char** p = (char**)malloc(8); *p = *(char**)d; return p; }
void* ql_any_copy_identity(void* d) { return d; }
void* ql_any_copy_nil(void* d) { (void)d; return 0; }



/* String.split(sep) → List<String>: same as Go strings.Split (empty sep → per UTF-8 character) */
static void ql_strlist_push(char*** buf, int* cap, int* n, const char* p, size_t len) {
    if (*n >= *cap) { *cap = *cap ? *cap * 2 : 4; *buf = (char**)realloc(*buf, sizeof(char*) * (size_t)(*cap)); }
    char* e = (char*)malloc(len + 1);
    memcpy(e, p, len); e[len] = 0;
    (*buf)[(*n)++] = e;
}
void* ql_str_split(const char* s, const char* sep) {
    char** buf = 0; int cap = 0, n = 0;
    size_t sn = strlen(sep);
    if (sn == 0) {
        const char* p = s;
        while (*p) {
            unsigned char c = (unsigned char)*p;
            int l = (c >= 0xF0) ? 4 : (c >= 0xE0) ? 3 : (c >= 0xC0) ? 2 : 1;
            ql_strlist_push(&buf, &cap, &n, p, (size_t)l);
            p += l;
        }
    } else {
        const char* p = s;
        for (;;) {
            const char* q = strstr(p, sep);
            if (!q) { ql_strlist_push(&buf, &cap, &n, p, strlen(p)); break; }
            ql_strlist_push(&buf, &cap, &n, p, (size_t)(q - p));
            p = q + sn;
        }
    }
    if (!buf) buf = (char**)malloc(sizeof(char*));
    struct ql_split_list { char** buf; int head; int tail; };
    struct ql_split_list* l = (struct ql_split_list*)calloc(1, sizeof(struct ql_split_list));
    l->buf = buf; l->head = 0; l->tail = n;
    return l;
}

/* List<String>: same layout as %ListS { i8** buf, i32 head, i32 tail }; toString same as the interpreter List.String() */
typedef struct { char** buf; int head; int tail; } ql_list_s;
char* ql_list_str_str(char** buf, int head, int tail) {
    size_t cap = 16, o = 0;
    char* r = (char*)malloc(cap);
    r[o++] = '[';
    for (int i = head; i < tail; i++) {
        const char* e = buf[i];
        size_t n = strlen(e);
        if (i > head) {
            if (o + 2 > cap) { cap *= 2; r = (char*)realloc(r, cap); }
            r[o++] = ','; r[o++] = ' ';
        }
        if (o + n + 2 > cap) { while (o + n + 2 > cap) cap *= 2; r = (char*)realloc(r, cap); }
        memcpy(r + o, e, n); o += n;
    }
    if (o + 2 > cap) { cap += 2; r = (char*)realloc(r, cap); }
    r[o++] = ']'; r[o] = 0;
    return r;
}

/* List<float>: same layout as %ListF { i64* buf, i32 head, i32 tail }, and each slot carries an element's
 * bit-cast double (the code generator stores it with `bitcast double to i64`), so the element is recovered
 * here with memcpy. Each element renders through ql_float_to_str, which is the one float spelling this
 * runtime has: a list and a bare float cannot disagree, and the interpreter's List.String() prints both the
 * same way. ql_float_to_str is defined by the generated module, which is why the caller also asks for it. */
char* ql_list_float_str(const long long* buf, int head, int tail) {
    size_t cap = 16, o = 0;
    char* r = (char*)malloc(cap);
    r[o++] = '[';
    for (int i = head; i < tail; i++) {
        double d;
        memcpy(&d, &buf[i], sizeof d);
        char* e = ql_float_to_str(d);
        size_t n = strlen(e);
        if (i > head) {
            if (o + 2 > cap) { cap *= 2; r = (char*)realloc(r, cap); }
            r[o++] = ','; r[o++] = ' ';
        }
        if (o + n + 2 > cap) { while (o + n + 2 > cap) cap *= 2; r = (char*)realloc(r, cap); }
        memcpy(r + o, e, n); o += n;
    }
    if (o + 2 > cap) { cap += 2; r = (char*)realloc(r, cap); }
    r[o++] = ']'; r[o] = 0;
    return r;
}

/* ================= HashTable<K,V> runtime =================
 * key = the interpreter-style structured string TypeName + ":" + String() (produced by ql_table_key);
 * value = the boxed interface{} value (data + RTTI descriptor), deep-copied via RTTI copy on Put. */
typedef struct ql_tent { char* key; void* data; void* rt; struct ql_tent* next; } ql_tent;
typedef struct { ql_tent** b; int cap; int len; } ql_table;
static unsigned ql_tbl_hash(const char* s) {
    unsigned h = 2166136261u;
    while (*s) { h ^= (unsigned char)*s++; h *= 16777619u; }
    return h;
}
void* ql_table_new(void) {
    ql_table* t = (ql_table*)calloc(1, sizeof(ql_table));
    t->cap = 64;
    t->b = (ql_tent**)calloc((size_t)t->cap, sizeof(ql_tent*));
    return t;
}
static ql_tent* ql_table_find(ql_table* t, const char* key) {
    ql_tent* e = t->b[ql_tbl_hash(key) % (unsigned)t->cap];
    while (e) { if (strcmp(e->key, key) == 0) return e; e = e->next; }
    return 0;
}
/* ql_table_key: turns a boxed key into the interpreter-style key string (TypeName:String()). */
char* ql_table_key(void* data, void* rt) {
    ql_rt* r = (ql_rt*)rt;
    const char* tn = r ? r->name : "nil";
    const char* sv = (r && data) ? r->str(data) : "nil";
    size_t n = strlen(tn) + strlen(sv) + 2;
    char* k = (char*)malloc(n);
    snprintf(k, n, "%s:%s", tn, sv);
    return k;
}
static void ql_table_grow(ql_table* t);
void ql_table_put(void* tp, char* key, void* data, void* rt) {
    ql_table* t = (ql_table*)tp;
    ql_rt* r = (ql_rt*)rt;
    void* cp = r ? (r->copy ? r->copy(data) : data) : 0;
    ql_tent* e = ql_table_find(t, key);
    if (e) { e->data = cp; e->rt = rt; return; }
    e = (ql_tent*)calloc(1, sizeof(ql_tent));
    e->key = key; e->data = cp; e->rt = rt;
    unsigned h = ql_tbl_hash(key) % (unsigned)t->cap;
    e->next = t->b[h]; t->b[h] = e; t->len++;
    if (t->len > t->cap * 2) ql_table_grow(t);
}
static void ql_table_grow(ql_table* t) {
    int nc = t->cap * 2;
    ql_tent** nb = (ql_tent**)calloc((size_t)nc, sizeof(ql_tent*));
    for (int i = 0; i < t->cap; i++) {
        ql_tent* e = t->b[i];
        while (e) {
            ql_tent* nx = e->next;
            unsigned h = ql_tbl_hash(e->key) % (unsigned)nc;
            e->next = nb[h]; nb[h] = e;
            e = nx;
        }
    }
    free(t->b); t->b = nb; t->cap = nc;
}
void* ql_table_get(void* tp, char* key, void** out_rt) {
    ql_table* t = (ql_table*)tp;
    ql_tent* e = ql_table_find(t, key);
    if (!e) { if (out_rt) *out_rt = 0; return 0; }
    if (out_rt) *out_rt = e->rt;
    return e->data;
}
int ql_table_contains(void* tp, char* key) { return ql_table_find((ql_table*)tp, key) != 0; }
void ql_table_remove(void* tp, char* key) {
    ql_table* t = (ql_table*)tp;
    unsigned h = ql_tbl_hash(key) % (unsigned)t->cap;
    ql_tent** pp = &t->b[h];
    while (*pp) {
        if (strcmp((*pp)->key, key) == 0) { *pp = (*pp)->next; t->len--; return; }
        pp = &(*pp)->next;
    }
}
int ql_table_size(void* tp) { return ((ql_table*)tp)->len; }
/* Key list (deterministic order: strcmp ascending; the interpreter keys() uses Go map order, itself nondeterministic). */
typedef struct { char** buf; int head; int tail; } ql_strlist;
static int ql_cmp_cstr(const void* a, const void* b) {
    return strcmp(*(char* const*)a, *(char* const*)b);
}
void* ql_table_keys(void* tp) {
    ql_table* t = (ql_table*)tp;
    int n = t->len;
    char** buf = (char**)malloc(sizeof(char*) * (size_t)(n > 0 ? n : 1));
    int k = 0;
    for (int i = 0; i < t->cap; i++)
        for (ql_tent* e = t->b[i]; e; e = e->next) {
            /* Keys() returns the **original value** of the key: a String key drops the "String:" prefix (same rule as the interpreter Keys) */
            buf[k++] = (strncmp(e->key, "String:", 7) == 0) ? (e->key + 7) : e->key;
        }
    if (n > 1) qsort(buf, (size_t)n, sizeof(char*), ql_cmp_cstr);
    ql_strlist* l = (ql_strlist*)calloc(1, sizeof(ql_strlist));
    l->buf = buf; l->head = 0; l->tail = n;
    return l;
}


/* Concrete V lookup: after confirming non-nil, returns the object pointer (for struct/List; missing key → an explicit runtime error) */
void* ql_any_ptr(void* d, void* rt) {
    if (!rt || !d) {
        fprintf(stderr, "error: TypeError: HashTable.get: missing key (nil value)\n");
        exit(1);
    }
    return d;
}
/* RTTI deep copy for List<int> (the visible range is copied into a new buffer) */
typedef struct { int* buf; int head; int tail; } ql_list_c;
void* ql_any_copy_list(void* d) {
    ql_list_c* src = (ql_list_c*)d;
    if (!src) return 0;
    int n = src->tail - src->head;
    if (n < 0) n = 0;
    int* buf = (int*)ql_alloc(sizeof(int) * (size_t)(n > 0 ? n : 1), 0);
    for (int i = 0; i < n; i++) buf[i] = src->buf[src->head + i];
    ql_list_c* cp = (ql_list_c*)ql_calloc(1, sizeof(ql_list_c), 0);
    cp->buf = buf; cp->head = 0; cp->tail = n;
    return cp;
}

char* ql_any_str_list(void* d) {
    struct ql_list_s { int* buf; int head; int tail; };
    struct ql_list_s* l = (struct ql_list_s*)d;
    return ql_list_int_str(l->buf, l->head, l->tail);
}
